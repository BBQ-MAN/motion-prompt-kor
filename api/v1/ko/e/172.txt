// 무엇이 일어나는가: 걷는 인물을 컷 없이 하나의 샷으로 계속 따라간다. 하단 바는 끊김 없는 한 줄로 유지된다.
// 동작 사양:
//   - 연결: 없음. 4.2초 동안 하나의 샷으로 인물을 계속 따라간다(#041 점프 컷은 다섯 개로 나뉜다)
//   - 피사체: 인물이 왼쪽 앞(x = -3.4m, 깊이 5.2m)에서 오른쪽 뒤(x = 2.8m, 깊이 9.6m)로 일정한 속도로 걷는다. 다리와 팔은 sin(t×7)×0.42rad로 흔들린다
//   - 카메라: 인물을 따라 오른쪽으로 0m → 1.4m 트럭하면서 방향을 돌린다. yaw = atan2(person x - camera x, depth)×0.8. 높이 1.4m
//   - 하단 바: "1ショット・カットなし"(원 샷, 컷 없음)라고 적힌 파란 바 하나와 재생 헤드. 끊김이 없음을 보여 준다
//   - 복귀: 4.2초에 멈춘 뒤 마지막 0.5초 동안 시작 상태로 크로스페이드(1회 4.7초)
//   - 오른쪽 위에 경과 시간 표시(NO CUT 0.0s → 4.2s)
// 쓰임새: 긴장감과 현장감을 끊지 않고 이어 갈 때, 일련의 움직임을 하나의 흐름으로 보여 줄 때.
// 태그: long take, oner, one shot, continuous, no cut, sequence shot, camera move
// 크기: 539x208 px · 1주기: 9.4 s
// Needs: kit camera2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/camera2.js
// 움직이는 견본: https://showheyohtaki.github.io/motion-prompt/#172
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 04
// 04 長回し: 歩く人物をカットなしで追い続ける1ショット。画面下の帯は1本のまま（#041 ジャンプカットのように割れない）
CELLS[3] = shot((k, w, h, u) => {
  const uu = Math.min(u, CYC - RESET);
  const draw = (kk, t) => {
    const p = t / (CYC - RESET);
    const x = lerp(-3.4, 2.8, p), z = lerp(5.2, 9.6, p), cx = lerp(0, 1.4, p);
    world(kk, w, h, cam({ x: cx, y: 1.4, yaw: Math.atan2(x - cx, z) * 0.8 }), { person: { x, z, walk: t * 7 }, fg: [[-2.6, 3.4], [3.6, 4.6, 'rock']] });
    strip(kk, w, h, [[0, CYC, '1ショット・カットなし', C.blue]], t);
  };
  const r = P(u, CYC - RESET, CYC);
  blend(k, kk => draw(kk, uu), kk => draw(kk, 0), eIOs(r));
  return `NO CUT ${(r > .5 ? 0 : uu).toFixed(1)}s`;
});
