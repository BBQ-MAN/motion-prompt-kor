// 익스트림 클로즈업 / Extreme close-up
// 팩: camera2 (카메라 워크와 화면 전환 2) · element 16 · 카테고리: 앵글·구도
// 무엇이 일어나는가: 얼굴 클로즈업에서 시작해 눈만으로 화면이 가득 찰 때까지 다가간다.
// 동작 사양:
//   - 카메라: 1.3m 앞 얼굴의 오른쪽 눈(중심에서 왼쪽 0.06m, 높이 1.66m)을 화면 중앙에 둔다. 카메라는 움직이지 않는다
//   - 초점 거리: 610px → 2040px(44mm → 148mm로 표시, 약 3.3배). f = 610 × (2040/610)^p. p는 0.5초부터 3.4초까지의 3차 ease-in-out(cubic-bezier(0.65,0,0.35,1)에 가까움)
//   - 얼굴: 파란 머리(반지름 0.16m)에 흰 눈(타원 0.045×0.028m), 어두운 눈동자(반지름 0.016m), 하이라이트, 흰 눈썹. 얼굴 전체 클로즈업(머리 폭 약 150px)에서 시작해 눈만 화면을 채우며 끝난다
//   - 배경: 나무와 건물은 3px 블러. 렌즈가 다가가면서 화면 밖으로 빠진다
//   - 복귀: 3.4~4.2초 정지 후 마지막 0.5초 동안 시작 상태로 크로스페이드. 오른쪽 위에 초점 거리와 약어 표시(44mm CU → 148mm ECU)
// 쓰임새: 감정이 가장 고조되는 순간, 작은 디테일 하나에 시선을 모을 때.
// 태그: extreme close-up, ecu, close-up, eye, detail, macro, framing
// 크기: 539x208 px · 1주기: 9.4 s
// Needs: kit camera2 -> https://bbq-man.github.io/motion-prompt-kor/api/v1/kit/camera2.js
// 움직이는 견본: https://bbq-man.github.io/motion-prompt-kor/#184
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 16
// 16 エクストリームクローズアップ: 顔のクローズアップから、目だけが画面を埋めるまで寄る
CELLS[15] = shot(camCell((k, w, h, p) => {
  const f = 610 * Math.pow(2040 / 610, p);
  world(k, w, h, cam({ f, x: -0.06, y: 1.66 }), {
    blurBg: 3, trees: true,
    extra: [[0, 0, 1.3, (kk) => {   // 目のある顔
      person(kk, 1);
      [[-0.06, 1.66], [0.06, 1.66]].forEach(([x, y]) => { kk.fillStyle = '#fff'; kk.beginPath(); kk.ellipse(x, y, 0.045, 0.028, 0, 0, 7); kk.fill(); kk.fillStyle = C.ink; kk.beginPath(); kk.arc(x + 0.008, y, 0.016, 0, 7); kk.fill(); kk.fillStyle = '#fff'; kk.beginPath(); kk.arc(x + 0.014, y + 0.007, 0.005, 0, 7); kk.fill(); kk.strokeStyle = '#fff'; kk.lineWidth = 0.012; kk.beginPath(); kk.arc(x, y + 0.02, 0.05, Math.PI * 1.15, Math.PI * 1.85); kk.stroke(); });
    }]]
  });
}, p => `${mm(610 * Math.pow(2040 / 610, p))} ${p > 0.85 ? 'ECU' : 'CU'}`));
