// 랙 포커스 / Rack focus
// 팩: camera (카메라 무빙과 화면 전환) · element 09 · 카테고리: 촬영 기법과 시간 연출
// 무엇이 일어나는가: 카메라는 움직이지 않고 초점을 앞쪽 꽃에서 뒤쪽 인물로 옮긴다. 흐려지는 쪽이 서로 바뀐다.
// 동작 사양:
//   - 카메라: 위치, 방향, 초점 거리 모두 고정(높이 1.1m). 바뀌는 것은 블러 양뿐
//   - 블러: 앞쪽 꽃(왼쪽 0.75m, 거리 2m)은 0px → 7px, 인물과 배경(인물은 오른쪽 0.9m, 거리 9m)은 6px → 0px. 지면 격자는 배경 블러의 0.6배
//   - 시간: 1.0초부터 2.6초까지 1.6초. 3차 ease-in-out(cubic-bezier(0.65,0,0.35,1) 상당)
//   - 초점 마커: 파란 네 모서리 브래킷이 꽃에서 인물로 직선 이동하며, 절반 폭이 46px → 30px로 줄어든다
//   - 피사체: 꽃도 인물도 움직이지 않는다
//   - 복귀: 2.6~4.2초는 정지, 마지막 0.5초 동안 시작 상태로 크로스페이드. 오른쪽 위에 초점 거리 표시(◎ 2m → ◎ 9m)
// 쓰임새: 컷을 나누지 않고 시선을 앞쪽에서 뒤쪽으로(또는 반대로) 옮길 때.
// 태그: rack focus, focus pull, pull focus, depth of field, blur, bokeh, foreground
// 크기: 539x208 px · 1주기: 9.4 s
// Needs: kit camera -> https://bbq-man.github.io/motion-prompt-kor/api/v1/kit/camera.js
// 움직이는 견본: https://bbq-man.github.io/motion-prompt-kor/#033
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 09
// 09 ラックフォーカス
CELLS[8] = shot(camCell((k, w, h, p) => {
  world(k, w, h, cam({ y: 1.1 }), {
    person: { x: 0.9, z: 9 }, blurFg: 7 * p, blurBg: 6 * (1 - p),
    extra: [[-0.75, 0, 2.0, (kk, s) => { kk.fillStyle = C.g3; kk.fillRect(-0.025, 0, 0.05, 0.75); kk.fillStyle = C.blueM; for (let i = 0; i < 6; i++) { kk.beginPath(); kk.ellipse(Math.cos(i) * 0.09, 0.8 + Math.sin(i) * 0.09, 0.08, 0.05, i, 0, 7); kk.fill(); } kk.fillStyle = C.blue; kk.beginPath(); kk.arc(0, 0.8, 0.05, 0, 7); kk.fill(); }, true]],
    after: (kk, cc) => {   // ピント位置の枠
      const a = proj(cc, -0.75, 0.8, 2.0), b = proj(cc, 0.9, 0.9, 9);
      const x = lerp(a.x, b.x, p), y = lerp(a.y, b.y, p), r = lerp(46, 30, p);
      kk.strokeStyle = C.blue; kk.lineWidth = 2; kk.beginPath();
      [[-1, -1], [1, -1], [-1, 1], [1, 1]].forEach(([sx, sy]) => { kk.moveTo(x + sx * r, y + sy * r - sy * 9); kk.lineTo(x + sx * r, y + sy * r); kk.lineTo(x + sx * r - sx * 9, y + sy * r); });
      kk.stroke();
    }
  });
}, p => `◎ ${lerp(2, 9, p).toFixed(0)}m`, 1.0, 2.6));
