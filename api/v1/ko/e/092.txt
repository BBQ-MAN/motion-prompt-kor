// 팩: lp (랜딩 페이지·웹사이트 연출) · element 20 · 카테고리: 로딩·페이지 이동·반복 연출
// 무엇이 일어나는가: 작품 목록에서 카드를 누르면 주황색 패널이 왼쪽에서 화면을 덮고, URL이 바뀐 뒤 패널이 오른쪽으로 빠지면 상세 페이지가 나온다. '목록으로'를 누르면 같은 방식으로 돌아간다.
// 동작 사양:
//   - 0.8초에 카드를 클릭. 0.9~1.3초에 패널(주황색, 뷰포트 전체, 가운데 로고)이 왼쪽에서 들어와 전체를 덮는다(easeInOut 3차)
//   - 1.3초, 덮여 있는 동안 페이지와 주소창의 URL(/works → /works/02)을 바꾼다
//   - 1.4~1.85초에 패널이 오른쪽으로 빠진다(easeInOut 3차)
//   - 3.0초에 '← 목록으로'를 클릭. 3.1~3.5초에 패널이 들어오고, 3.5초에 페이지가 바뀌고, 3.6~4.05초에 빠진다
//   - 패널은 헤더를 포함해 뷰포트 전체를 덮는다
// 쓰임새: 브랜드 사이트나 포트폴리오에서 페이지와 페이지 사이를 자연스럽게 잇고 싶을 때.
// 태그: page-transition, transition, view-transitions, navigation, wipe, barba, spa
// 크기: 539x208 px · 1주기: 4.8 s
// Needs: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 움직이는 견본: https://showheyohtaki.github.io/motion-prompt/#092
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 20
// 20 ページ遷移アニメーション（リンクを押すと、色の幕が横切って次のページに切り替わる）
CELLS[19] = (k, w, h, u) => {
  frame(k, u >= 1.3 && u < 3.5 ? 'example.com/works/02' : 'example.com/works');
  const CLK = [0.8, 3.0];
  const cover = t0 => eIO(P(u, t0, t0 + 0.4)), uncover = t0 => eIO(P(u, t0, t0 + 0.45));
  // 幕の左端: 入る −VW→0、出る 0→VW
  const xs = u < 1.4 ? -VW * (1 - cover(0.9)) : u < 3.1 ? VW * uncover(1.4) : u < 3.6 ? -VW * (1 - cover(3.1)) : VW * uncover(3.6);
  const pageB = u >= 1.3 && u < 3.5;
  const [cx, cy] = path([[0, 420, 150], [0.65, 255, 100], [2.6, 255, 100], [2.85, 46, 50], [3.3, 46, 50], [4.6, 420, 150], [4.8, 420, 150]], u);
  view(k, () => {
    header(k, 0);
    if (!pageB) {
      txt(k, 'WORKS', 34, 62, `800 18px ${EN}`, C.ink, 'left', 'alphabetic', '2px');
      for (let i = 0; i < 3; i++) { const x = 34 + i * 152; pic(k, x, 74, 136, 60, i, 6); bars(k, x, 142, 100, 1, C.ink, 7); }
    } else {
      txt(k, '← 一覧へ', 34, 55, `700 12px ${JP}`, C.sub, 'left', 'middle');
      pic(k, 34, 68, 240, 90, 1, 8);
      txt(k, 'Project 02', 296, 96, `800 22px ${EN}`, C.ink); bars(k, 296, 110, 180, 3);
    }
    if (xs > -VW && xs < VW) { k.fillStyle = C.acc; k.fillRect(xs, 0, VW, VH); logo(k, xs + VW / 2 - 40, VH / 2, '#fff'); }
    click(k, cx, cy, u, CLK);
  });
};
