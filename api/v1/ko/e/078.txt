// 스크롤 스냅 / Scroll snap
// 팩: lp (랜딩 페이지·웹사이트 연출) · element 06 · 카테고리: 스크롤 연출
// 무엇이 일어나는가: 조금만 스크롤하면 나머지는 자동으로 넘어가, 다음 섹션의 위쪽 끝이 화면 위에 딱 맞는다.
// 동작 사양:
//   - 섹션 높이 = 뷰포트 높이(168px). 첫 번째는 옅은 파란 바탕, 두 번째는 남색
//   - 2.4초에 한 번. 0.45~0.85초에 사용자가 34px만 스크롤(easeOut 3차)
//   - 0.9~1.5초에 남은 134px를 자동으로 이동해 다음 섹션 위쪽 끝에 맞춘다(easeOut 3차)
//   - 뒤의 2.4초는 같은 동작을 위쪽으로 해서 첫 번째 섹션으로 돌아간다(scroll-snap-type: y mandatory의 동작)
// 쓰임새: 한 화면씩 끊어 보여 주는 랜딩 페이지나 슬라이드 같은 페이지를 만들고 싶을 때.
// 태그: scroll-snap, snap, full-page, sections, css, mandatory, scroll
// 크기: 539x208 px · 1주기: 4.8 s
// Needs: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 움직이는 견본: https://showheyohtaki.github.io/motion-prompt/#078
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 06
// 06 スクロールスナップ（少しスクロールすると、次のセクションの頭までピタッと送られる）
CELLS[5] = (k, w, h, u) => {
  frame(k);
  const T = CYC / 2, lu = mod(u, T), dir = u < T ? 1 : -1, from = u < T ? 0 : VH, DRIFT = 34;
  const y = from + dir * (DRIFT * eOut(P(lu, 0.45, 0.85)) + (VH - DRIFT) * eOut(P(lu, 0.9, 1.5)));
  view(k, () => {
    k.save(); k.translate(0, -y);
    for (let i = 0; i < 2; i++) {
      const y0 = i * VH, dark = i === 1;
      k.fillStyle = dark ? C.ink : C.blueL; k.fillRect(0, y0, VW, VH);
      txt(k, '0' + (i + 1), 34, y0 + 70, `800 46px ${EN}`, C.acc);
      txt(k, dark ? '使ってみる' : '知ってもらう', 34, y0 + 102, `800 18px ${JP}`, dark ? '#fff' : C.ink);
      bars(k, 34, y0 + 118, 170, 2, dark ? 'rgba(255,255,255,.3)' : '#C3CDEA');
      pic(k, 262, y0 + 24, 220, 120, dark ? 2 : 0, 8);
    }
    k.restore();
    sbar(k, y / VH, 0.5);
  });
};
