// 골든 아워 / Golden hour
// 팩: photo (사진 표현·빛과 스타일) · element 07 · 카테고리: 조명·시간대
// 무엇이 일어나는가: 해가 지평선 바로 위까지 내려오면서 하늘과 땅이 주황으로 물들고, 인물의 왼쪽이 빛나며, 가로등과 나무의 그림자가 오른쪽으로 길게 늘어난다.
// 동작 사양:
//   - 장면 1: 지평선(y 150) 앞에 가슴 위까지 보이는 인물(머리 중심 x 270, y 78, 반지름 29px, 얼굴·머리카락 없음).
//   - 해: 반지름 18px, (x 110, y −30) → (x 96, y 136). 언덕에 반쯤 가려진다
//   - 색: 하늘 #FAF9F7 → 위쪽 #FFE8BC, 지평선 근처 #FFCF80, 땅 #F1F0EE → #F6DDB0. 인물의 빛 #FFB547, 그림자 #3E3650
//   - 머리: 빛은 왼쪽 약 66°에서 오고 경계보다 오른쪽은 그림자. 몸통은 왼쪽이 빛, 오른쪽이 그림자
//   - 그림자: 가로등과 나무 밑동에서 오른쪽으로, 길이 0 → 150px(#DDAA6E). 오른쪽 위 시각 12:00 → 17:40
//   - 시간: 0.4~1.4초에 효과가 들어오고 4.0초까지 유지된 뒤 4.8초까지 원래대로 돌아간다(1회 5초, 3차 이즈 인 아웃(cubic-bezier(0.65,0,0.35,1) 상당))
// 쓰임새: 따뜻하고 아련하며 부드러운 분위기의 사진을 만들 때.
// 태그: golden hour, magic hour, sunset light, warm light, long shadows, low sun
// 크기: 539x208 px · 1주기: 5 s
// Needs: kit photo -> https://bbq-man.github.io/motion-prompt-kor/api/v1/kit/photo.js
// 움직이는 견본: https://bbq-man.github.io/motion-prompt-kor/#103
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 07
// ゴールデンアワー: 太陽が地平線近くまで下り、全体が橙に染まって影が長く伸びる
CELLS[6] = cell((k, u) => {
  const a = env(u), R = PR, W = torsoW(R);
  const GOLD = {
    sky: '#FFE8BC', sky2: '#FFCF80', far: '#F2C58C', mid: '#E8B27A', ground: '#F6DDB0', ground2: '#ECC892', mark: '#F6DDB0', cast: '#DDAA6E',
    bld: '#F7DCAE', edge: '#D9AE76', win: '#ECC892', winOff: '#ECC892', tree: '#E9C08A', trunk: '#C9965E', post: '#B98A58', bulb: '#C9965E',
    person: '#5C5470', shade: '#3E3650', lit: '#FFB547', car: '#A08E8E', glass: '#F6DDB0', tire: '#3E3650', sun: '#F59A24', glow: '#FFD48A'
  };
  const p = pal(BASE, GOLD, a);
  stagePortrait(k, p, { sun: [lerp(110, 96, a), lerp(-30, 136, a), 18, 1], cast: lerp(0, 150, a) });
  const lit = mixc(p.person, GOLD.lit, a), sh = mixc(p.person, GOLD.shade, a);
  bust(k, PX, PY, R, p.person, kk => {
    kk.fillStyle = lit; kk.beginPath(); kk.arc(PX, PY, R, 0, TAU); kk.fill();
    headShade(kk, PX, PY, R, 1.15, 1, sh);
    torso(kk, PX, PY, R, PX - W, PX - 0.5 * W, lit);
    torso(kk, PX, PY, R, PX + 0.4 * W, PX + W, sh);
  });
  hud(k, clock(lerp(720, 1060, a)));
});
