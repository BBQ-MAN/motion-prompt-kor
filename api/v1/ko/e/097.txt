// 역광 / Backlighting
// 팩: photo (사진 표현·빛과 스타일) · element 01 · 카테고리: 조명·시간대
// 무엇이 일어나는가: 화면 밖에 있던 해가 인물의 머리 바로 뒤로 옮겨 온다. 인물 윤곽에 가는 빛 테두리가 생기고, 화면 전체에 옅고 따뜻한 빛이 깔리며, 카메라 쪽 몸은 어둡게 가라앉는다.
// 동작 사양:
//   - 장면 1: 지평선(y 150) 앞에 가슴 위까지 보이는 인물(머리 중심 x 270, y 78, 반지름 29px, 얼굴·머리카락 없음). 빛은 앰버 #FFC75A의 평평한 면, 그림자는 평평한 #33405A 면
//   - 해: 반지름 20px. 왼쪽 위 화면 밖(x 40, y −40)에서 머리 바로 뒤(x 270, y 56)로 이동하고, 주위에 반지름 2.6배·1.7배의 평평한 빛 고리(#FFE3A3, 35%·60%)
//   - 윤곽: 인물 형태를 7px 선으로 그린 뒤 안을 채워, 바깥쪽 3.5px의 밝은 테두리만 남긴다
//   - 인물 #4D5A75 → #33405A, 하늘 #FAF9F7 → #FFF4DC, 화면 전체에 #FFE3A3를 22% 겹친다
//   - 시간: 0.4~1.4초에 효과가 들어오고 4.0초까지 유지된 뒤 4.8초까지 원래대로 돌아간다(1회 5초, 3차 이즈 인 아웃(cubic-bezier(0.65,0,0.35,1) 상당))
// 쓰임새: 산뜻하고 빛나는 인물 사진이나, 빛에 감싸인 분위기를 내고 싶을 때.
// 태그: backlighting, backlit, contre-jour, sun behind, haze, glow, halo
// 크기: 539x208 px · 1주기: 5 s
// Needs: kit photo -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/photo.js
// 움직이는 견본: https://showheyohtaki.github.io/motion-prompt/#097
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 01
// 逆光: 太陽が画面の外から頭の真後ろへ回り込み、輪郭が光って人物は暗くなる
CELLS[0] = cell((k, u) => {
  const a = env(u);
  const BACK = Object.assign({}, BASE, { sky: '#FFF4DC', sky2: '#FFEBC4', far: '#F6E2BE', ground: '#F7EAD3', bld: '#F7E9D2', winOff: '#EEDCBF', win: '#EEDCBF', tree: '#EEDCBF', edge: '#E3CFAE', post: '#CDB894', trunk: '#E3CFAE', bulb: '#E3CFAE' });
  stagePortrait(k, pal(BASE, BACK, a), { sun: [lerp(40, PX, a), lerp(-40, PY - 22, a), 20, 1] });
  // 輪郭の光: 胸像の形を太い線でなぞってから中を塗り、縁だけを残す
  k.save(); k.globalAlpha = a; bustPath(k, PX, PY, PR); k.strokeStyle = BASE.lit; k.lineWidth = 7; k.lineJoin = 'round'; k.stroke(); k.restore();
  bust(k, PX, PY, PR, mixc(BASE.person, BASE.shade, a));
  // かすみ: 全体に光の色を薄く重ねる
  k.save(); k.globalAlpha = 0.22 * a; k.fillStyle = BASE.glow; k.fillRect(0, 0, CW, CH); k.restore();
});
