// 팩: ui (UI 컴포넌트 이름) · element 22 · 카테고리: 내비게이션·상태 표시
// 무엇이 일어나는가: "전송"을 누르면 카드가 하얗게 덮이고 가운데에서 파란 호가 돈다. 끝나면 체크로 바뀌고 사라진다.
// 동작 사양:
//   - 4.5초 주기, 카드 542×191. 커서가 1.0초에 아래 가운데 "전송"(120×40)을 클릭한다.
//   - 1.05–1.2초에(linear) 카드 전체에 흰 스크림(불투명도 0.78)과 스피너가 페이드인된다.
//   - 스피너: (271, 92)에 반경 26, 두께 6의 연한 링과 그 위의 파란 243°(1.35π) 호(끝은 둥글게). 초당 1.3회전(468°/s).
//   - 2.95–3.05초에 스피너가 사라지고, 2.95–3.2초에 흰 체크가 든 파란 원(반경 28)이 배율 0.7 → 1(easeOutCubic)로 나타난다.
//   - 3.75–4.0초에 모두 linear로 사라진다.
// 쓰임새: 전송이나 저장처럼 끝나는 시점을 알 수 없는 짧은 대기에.
// 태그: spinner, loading, loader, activity-indicator, progress-circle, busy
// 크기: 542x191 px · 1주기: 4.5 s
// Needs: kit ui -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/ui.js
// 움직이는 견본: https://showheyohtaki.github.io/motion-prompt/#070
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 22
// 22 スピナー（Spinner）
CELLS[21] = ui((w, h, t) => {
  const load = seg(t, 1.05, 1.2) * (1 - seg(t, 2.95, 3.05));
  const done = seg(t, 2.95, 3.15) * (1 - seg(t, 3.75, 4.0));
  pageLines(24, 18, 494); strokeRR(24, 90, 494, 34, 7, C.line2, 1.5);
  fillRR(w / 2 - 60, 140, 120, 40, 8, C.blue); text('送信', w / 2, 160, 16, '#fff', 600, 'center');
  const ov = Math.max(load, done);
  if (ov > 0) {
    X.fillStyle = `rgba(255,255,255,${.78 * ov})`; X.fillRect(0, 0, w, h);
    if (load > 0) { X.save(); X.globalAlpha = load; ring(w / 2, 92, 26, '#e3e8f5', 6); X.beginPath(); const a0 = t * Math.PI * 2 * 1.3; X.arc(w / 2, 92, 26, a0, a0 + Math.PI * 1.35); X.strokeStyle = C.blue; X.lineWidth = 6; X.lineCap = 'round'; X.stroke(); X.restore(); }
    if (done > 0) { X.save(); X.globalAlpha = done; const s = .7 + .3 * eo(seg(t, 2.95, 3.2)); X.translate(w / 2, 92); X.scale(s, s); circle(0, 0, 28, C.blue); check(0, 0, 11, '#fff', 4); X.restore(); }
  }
  pointer([[.3, w + 30, h + 30], [.95, w / 2 + 6, 166], [1.5, w / 2 + 6, 166], [2.2, w + 30, h + 30]], [1.0], t);
});
