// 동작 사양:
//   - 둥근 막대(24×60, 모서리 반경 12)를 중앙의 반지름 58 원 위에 최대 12개 배치한다. 각각 바깥을 향하게 돌린다(각도 + 90°).
//   - 개수: 0.35–1.7초에 1→12, 3.6–4.45초에 12→1(2차 ease-in-out). 새 복제본은 이전 자리에서 30° 앞의 자기 자리로 미끄러져 나온다(3차 ease-out).
//   - 전체를 4.7초에 한 바퀴 돌린다. 각 복제본의 크기를 1 ± 0.22로 4.7초에 2번 맥동시키고, 하나 앞으로 갈수록 0.55rad씩 늦춘다(고리를 도는 물결이 된다).
//   - 색은 파랑, 코럴, 노랑, 민트 순. 원본 1개에는 흰 테두리. 오른쪽 위에 개수를 나타내는 "×N".
// 쓰임새: 로딩, 배경 패턴, 꽃이나 빛의 방사처럼 같은 모양을 반복해 늘어놓고 싶을 때.
// 태그: repeater, duplicate, radial, array, clone, pattern, loader
// 크기: 539x208 px · 1주기: 4.7 s
// Needs: kit motion2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/motion2.js
// 움직이는 견본: https://showheyohtaki.github.io/motion-prompt/#134
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 14
// 14 リピーター（1つの図形が複製されて円に並び、少しずつずれながら回る）
CELLS[13] = (k, w, h, u) => {
  u = snapU(u);
  stage(k);
  const N = 12, X = w / 2, Y = h / 2, R = 58, spin = 2 * Math.PI * u / CYC;
  const cnt = 1 + (N - 1) * (eIOs(P(u, 0.35, 1.7)) - eIOs(P(u, 3.6, 4.45)));
  for (let i = N - 1; i >= 0; i--) {
    const vis = i === 0 ? 1 : clamp(cnt - i); if (vis <= 0) continue;
    const ang = spin + (i === 0 ? 0 : (i - 1 + eOut(vis))) * 2 * Math.PI / N;
    const pul = (1 + 0.22 * Math.sin(2 * Math.PI * 2 * u / CYC - i * 0.55)) * (0.55 + 0.45 * vis);
    k.save(); k.translate(X + Math.cos(ang) * R, Y + Math.sin(ang) * R); k.rotate(ang + Math.PI / 2); k.scale(pul, pul);
    k.fillStyle = PAL[i % 4]; rr(k, -12, -30, 24, 60, 12); k.fill();
    if (i === 0) { k.strokeStyle = '#fff'; k.lineWidth = 4; rr(k, -12, -30, 24, 60, 12); k.stroke(); }
    k.restore();
  }
  txt(k, '×' + Math.round(cnt), w - 34, 46, `800 40px ${DISP}`, C.yel, 'right', 'middle');
};
