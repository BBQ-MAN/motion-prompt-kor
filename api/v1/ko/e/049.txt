// 무엇이 일어나는가: "열기" 버튼을 누르면 화면 전체가 어두워지고 가운데에 작은 창이 살짝 커지며 나타난다. ×를 누르면 닫힌다.
// 동작 사양:
//   - 4.5초 주기, 카드 542×191. 커서가 0.95초에 왼쪽 아래 "열기" 버튼(120×36)에 닿고 1.0초에 클릭한다.
//   - 열림 정도 a: 1.0–1.32초에 easeOutCubic으로 0→1. 카드 전체에 rgba(18,20,28) 스크림을 불투명도 0.42·a로 깐다.
//   - 창: 흰색 280×140, 모서리 반경 12, 중심 (271, 97). 그림자 blur 24, y 오프셋 8, 알파 0.25. 배율 0.9 + 0.1·a, 불투명도 a.
//   - 창 안: 굵은 제목 막대, 본문 2줄, 오른쪽 아래 파란 버튼 88×30, 오른쪽 위 ×.
//   - 닫기: 커서가 3.0초까지 ×로 이동해 3.3초에 클릭한다. a에 (1 − easeInOutCubic(3.3–3.58초))를 곱해 사라지게 한다.
// 쓰임새: 확인이나 짧은 입력처럼 하던 작업을 멈추고 한 가지에 집중시켜야 할 때.
// 태그: modal, modal-dialog, dialog, popup, overlay, backdrop, lightbox, confirm
// 크기: 542x191 px · 1주기: 4.5 s
// Needs: kit ui -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/ui.js
// 움직이는 견본: https://showheyohtaki.github.io/motion-prompt/#049
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 01
// 01 モーダル（Modal dialog）
CELLS[0] = ui((w, h, t) => {
  pageLines(24, 18, 494);
  fillRR(24, 92, 120, 36, 8, C.blue); text('開く', 84, 110, 16, '#fff', 600, 'center');
  fillRR(170, 92, 348, 36, 8, C.block); fillRR(24, 142, 494, 38, 8, C.block);
  const a = eo(seg(t, 1.0, 1.32)) * (1 - eio(seg(t, 3.3, 3.58)));
  if (a > 0) {
    X.fillStyle = `rgba(18,20,28,${.42 * a})`; X.fillRect(0, 0, w, h);
    X.save(); X.globalAlpha = a; X.translate(271, 97); const s = .9 + .1 * a; X.scale(s, s);
    shadowed(() => fillRR(-140, -70, 280, 140, 12, '#fff'), 24, 8, .25);
    bar(-116, -48, 120, C.ink, 11); bar(-116, -20, 232); bar(-116, -2, 180);
    fillRR(28, 26, 88, 30, 7, C.blue);
    X.strokeStyle = '#9a9896'; X.lineWidth = 2.2; X.lineCap = 'round'; X.beginPath(); X.moveTo(110, -54); X.lineTo(120, -44); X.moveTo(120, -54); X.lineTo(110, -44); X.stroke();
    X.restore();
  }
  pointer([[.3, w + 30, h + 30], [.95, 86, 114], [1.5, 86, 114], [3.0, 389, 50], [3.5, 389, 50], [4.2, w + 30, h + 30]], [1.0, 3.3], t);
});
