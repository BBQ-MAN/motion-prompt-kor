// 무엇이 일어나는가: 그래프 막대가 왼쪽부터 조금씩 시간 차를 두고 차례로 올라온다.
// 동작 사양:
//   - 2.35초에 1회. 막대 7개(폭 44, 간격 20, 높이 52, 84, 68, 112, 92, 136, 158). 색은 파랑, 코랄, 노랑, 민트 순으로 반복.
//   - i번째 막대(0부터 셈)는 0.15 + 0.12·i초에 시작해 0.5초 동안 높이 0→100%로 자란다. backOut(s = 2.6), 밑변 고정, 넘쳤다가 자리 잡는다.
//   - 막대 위 흰 점(반지름 6.5, 막대 윗면에서 16px 위): 막대 시작 0.34초 뒤 스프링 1 − e^(−7t)·cos(16t)로 나타난다.
//   - 퇴장: 막대는 1.78 + 0.045·i초부터 0.26초 동안 3차 easeIn으로 줄어든다. 점은 그보다 먼저 1.70 + 0.045·i초부터 0.2초 동안 사라진다.
// 쓰임새: 리스트·그래프·카드처럼 여러 요소를 순서대로 보여 주며 시선을 이끌 때.
// 태그: stagger, sequence, delay, cascade, bar-chart, list, entrance
// 크기: 539x208 px · 1주기: 4.7 s
// Needs: kit motion -> https://bbq-man.github.io/motion-prompt-kor/api/v1/kit/motion.js
// 움직이는 견본: https://bbq-man.github.io/motion-prompt-kor/#002
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 02
// 02 スタガー（グラフの棒が、少しずつ時間をずらして順に立ち上がる）
CELLS[1] = (k, w, h, u) => {
  stage(k);
  const N = 7, BW = 44, G = 20, x0 = (w - (N * BW + (N - 1) * G)) / 2, base = h - 30;
  const HS = [52, 84, 68, 112, 92, 136, 158];
  k.fillStyle = C.navy3; rr(k, x0 - 18, base, N * BW + (N - 1) * G + 36, 5, 2.5); k.fill();
  const lu = mod(u, CYC / 2);
  for (let i = 0; i < N; i++) {
    const t0 = 0.15 + i * 0.12, g = backOut(P(lu, t0, t0 + 0.5), 2.6) * (1 - eIn(P(lu, 1.78 + i * 0.045, 2.04 + i * 0.045)));
    if (g <= 0.001) continue;
    const hh = HS[i] * g, x = x0 + i * (BW + G);
    k.fillStyle = PAL[i % 4]; rr(k, x, base - hh, BW, hh, [12, 12, 3, 3]); k.fill();
    // 棒の上の点（さらに遅れて出る）
    const d = spring(lu - t0 - 0.34, 16, 7) * (1 - eIn(P(lu, 1.7 + i * 0.045, 1.9 + i * 0.045)));
    if (d > 0.01) { k.fillStyle = '#fff'; k.beginPath(); k.arc(x + BW / 2, base - hh - 16, 6.5 * d, 0, 7); k.fill(); }
  }
};
