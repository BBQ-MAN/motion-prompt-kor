// 마그네틱 버튼 / Magnetic button
// 팩: lp (랜딩 페이지·웹사이트 연출) · element 12 · 카테고리: 호버·클릭 인터랙션
// 무엇이 일어나는가: 커서가 다가오면 버튼이 커서 쪽으로 끌려가 따라붙고, 커서가 벗어나면 출렁이며 제자리로 돌아온다.
// 동작 사양:
//   - 버튼 168×52(주황색, 모서리 반경 26)를 가운데에 둔다. 끌어당기는 범위는 버튼 중심에서 반지름 110px
//   - 범위 안: 목표 오프셋 = (커서 − 버튼 중심) × 0.35. 매 프레임(60fps) 목표까지 남은 거리의 18%씩 다가간다
//   - 범위를 벗어나면 그때의 오프셋에서 0으로 1.1초 동안 돌아온다. 이징은 elastic.out(1, 0.3)(2^(−10x)·sin((x − 0.075)·2π/0.3) + 1)
//   - 커서는 왼쪽 바깥에서 들어와 버튼 주위를 돈 뒤 오른쪽 바깥으로 나간다
// 쓰임새: 가입이나 문의처럼 가장 눌러 주길 바라는 버튼을 돋보이게 하고 싶을 때.
// 태그: magnetic, magnetic-button, hover, cursor, elastic, cta, interaction
// 크기: 539x208 px · 1주기: 4.8 s
// Needs: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 움직이는 견본: https://showheyohtaki.github.io/motion-prompt/#084
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 12
// 12 マグネティックボタン（カーソルが近づくと、ボタンが吸い寄せられ、離れるとぷるんと戻る）
CELLS[11] = (k, w, h, u) => {
  frame(k);
  const BX = VW / 2, BY = 84, R = 110, PULL = 0.35;
  const cur = t => path([[0, -20, 130], [0.9, 170, 104], [1.5, 300, 62], [2.1, 318, 110], [2.7, 205, 66], [3.1, 250, 92], [3.6, 545, 120], [4.8, 545, 120]], t);
  const inside = t => { const [x, y] = cur(t); return Math.hypot(x - BX, y - BY) < R; };
  // 範囲に入った時刻から 60fps で目標（カーソルとの差 × 0.35）へ毎フレーム 18% ずつ近づく
  const pull = t => {
    let t0 = t; for (let i = 0; i < 600 && inside(t0 - 1 / 120); i++) t0 -= 1 / 120;
    let ox = 0, oy = 0; for (let s = t0; s <= t; s += 1 / 60) { const [x, y] = cur(s); ox += ((x - BX) * PULL - ox) * 0.18; oy += ((y - BY) * PULL - oy) * 0.18; }
    return [ox, oy];
  };
  let ox = 0, oy = 0;
  if (inside(u)) [ox, oy] = pull(u);
  else {
    let t1 = u; for (let i = 0; i < 600 && !inside(t1); i++) t1 -= 1 / 120;
    if (inside(t1)) { const [lx, ly] = pull(t1), e = 1 - eEl(P(u, t1, t1 + 1.1)); ox = lx * e; oy = ly * e; }
  }
  view(k, () => {
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH);
    k.save(); k.translate(BX + ox, BY + oy);
    k.shadowColor = 'rgba(255,90,44,.35)'; k.shadowBlur = 16; k.shadowOffsetY = 6;
    k.fillStyle = C.acc; rr(k, -84, -26, 168, 52, 26); k.fill(); k.shadowColor = 'transparent';
    txt(k, 'お問い合わせ', 0, 1, `800 16px ${JP}`, '#fff', 'center', 'middle');
    k.restore();
    const [cx, cy] = cur(u); cursor(k, cx, cy);
  });
};
