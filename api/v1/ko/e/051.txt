// 팩: ui (UI 컴포넌트 이름) · element 03 · 카테고리: 열리고 겹치는 컴포넌트
// 무엇이 일어나는가: 손가락으로 "상세 보기"를 누르면 아래에서 패널이 올라오고, 위쪽 핸들을 아래로 끌면 닫힌다.
// 동작 사양:
//   - 4.5초 주기, 카드 542×191. 손가락(반경 17의 반투명 원, 흰 테두리)이 1.0초에 아래 가운데 "상세 보기" 알약 버튼(170×36)을 누른다.
//   - 열림 정도: 1.05–1.45초에 easeOutCubic으로 0→1. 카드 전체에 rgba(18,20,28) 스크림을 불투명도 0.36·a로 깐다.
//   - 시트: 흰색, 높이 136, 위쪽 모서리 반경 18, 위 가장자리 y = 191 − 136·a. 그림자 blur 22, y 오프셋 −4. 위 가운데에 52×6 핸들.
//   - 시트 안: 굵은 제목 막대와 26×26 아이콘이 달린 2줄.
//   - 닫기: 손가락이 2.75초에 핸들 (271, 68)에 닿고 3.0–3.6초에 아래로 끈다. a에 (1 − easeInOutCubic(3.0–3.6초))를 곱해 시트가 손가락을 따라 아래로 사라진다.
// 쓰임새: 모바일에서 지금 화면을 그대로 둔 채 상세 정보나 선택지를 보여 줄 때.
// 태그: bottom-sheet, sheet, modal-sheet, mobile, slide-up, drag-to-dismiss, panel
// 크기: 542x191 px · 1주기: 4.5 s
// Needs: kit ui -> https://bbq-man.github.io/motion-prompt-kor/api/v1/kit/ui.js
// 움직이는 견본: https://bbq-man.github.io/motion-prompt-kor/#051
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 03
// 03 ボトムシート（Bottom sheet）
CELLS[2] = ui((w, h, t) => {
  pageLines(24, 16, 494); fillRR(24, 76, 239, 54, 8, C.block); fillRR(279, 76, 239, 54, 8, C.block);
  fillRR(186, 144, 170, 36, 18, C.blue); text('詳細を見る', 271, 162, 15, '#fff', 600, 'center');
  const open = eo(seg(t, 1.05, 1.45));
  const drag = eio(seg(t, 3.0, 3.6));
  const a = open * (1 - drag);
  if (open > 0 && a > 0.001) {
    X.fillStyle = `rgba(18,20,28,${.36 * a})`; X.fillRect(0, 0, w, h);
    const sh = 136, sy = h - sh * a;
    shadowed(() => { rr(0, sy, w, sh + 20, 18); X.fillStyle = '#fff'; X.fill(); }, 22, -4, .2);
    fillRR(w / 2 - 26, sy + 10, 52, 6, 3, C.line2);
    bar(28, sy + 30, 160, C.ink, 11);
    for (let i = 0; i < 2; i++) { const y = sy + 56 + i * 36; fillRR(28, y, 26, 26, 6, i ? C.mb : C.blue); bar(66, y + 4, 220, C.line); bar(66, y + 17, 140, C.line, 7); }
  }
  const sy = h - 136 * a;
  finger([[.3, 271, 162], [1.6, 271, 162], [2.75, w / 2, h - 136 + 13], [3.0, w / 2, h - 136 + 13], [3.6, w / 2, h + 10]], [1.0, 3.0], t, (t > .6 && t < 1.6) || (t > 2.75 && t < 3.6));
});
