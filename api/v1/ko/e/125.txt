// 팩: motion2 (모션 그래픽 연출 2탄) · element 05 · 카테고리: 모션 테크닉
// 무엇이 일어나는가: 하트가 심장 박동처럼 두근두근 크기를 키웠다 줄이고, 고리가 퍼져 나간다.
// 동작 사양:
//   - 코럴색 하트. 1.175초에 1박(4.7초에 4박). 1박은 두 번 뛰는 '두근두근'.
//   - 크기 = 1 + 0.26·G(0.2초, 폭 0.065) + 0.15·G(0.44초, 폭 0.07). G = e^(−((t − 중심) ÷ 폭)²)(빠르게 부풀었다가 바로 돌아오는 봉우리).
//   - 하트 윤곽선 2개를 퍼뜨린다: 0.16초와 0.40초부터 0.8초 동안 크기 1.2→2.4(3차 ease-out).
//   - 고리는 퍼지면서 불투명도 0.75→0(두 번째는 60% 농도), 선 두께 8.5→1.5px.
// 쓰임새: 좋아요, 녹음 중, 라이브 표시, 눌러 주길 바라는 버튼처럼 살아 있는 느낌이나 주목을 끌고 싶을 때.
// 태그: pulse, heartbeat, throb, scale, loop, like, attention
// 크기: 539x208 px · 1주기: 4.7 s
// Needs: kit motion2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/motion2.js
// 움직이는 견본: https://showheyohtaki.github.io/motion-prompt/#125
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 05
// 05 パルス（ハートが鼓動のように、ドクン・ドクンと大きさを脈打たせる）
CELLS[4] = (k, w, h, u) => {
  stage(k);
  const T = CYC / 4, lu = mod(u, T), X = w / 2, Y = h / 2 + 10;
  const g = (c, s) => Math.exp(-Math.pow((lu - c) / s, 2));
  const b = 0.26 * g(0.2, 0.065) + 0.15 * g(0.44, 0.07) + 0.26 * Math.exp(-Math.pow((lu - T - 0.2) / 0.065, 2));
  for (const [t0, s] of [[0.16, 1], [0.4, 0.6]]) {
    const q = P(lu, t0, t0 + 0.8); if (q <= 0 || q >= 1) continue;
    k.save(); k.translate(X, Y); const sc = 1.2 + 1.2 * eOut(q); k.scale(sc, sc); heartPath(k, 44);
    k.strokeStyle = C.coral; k.globalAlpha = (1 - q) * 0.75 * s; k.lineWidth = (7 * (1 - q) + 1.5) / sc; k.stroke(); k.restore();
  }
  k.globalAlpha = 1;
  k.save(); k.translate(X, Y); k.scale(1 + b, 1 + b); heartPath(k, 44); k.fillStyle = C.coral; k.fill();
  k.fillStyle = 'rgba(255,255,255,.75)'; k.beginPath(); k.ellipse(-30, -28, 9, 6, -0.7, 0, 7); k.fill(); k.restore();
};
