// 팩: lp (랜딩 페이지·웹사이트 연출) · element 18 · 카테고리: 로딩·페이지 이동·반복 연출
// 무엇이 일어나는가: 스크롤로 화면에 들어온 회색 플레이스홀더 안에서 스피너가 돌고, 이미지가 도착하는 대로 페이드인된다. 윗줄은 처음부터 불러온 상태다.
// 동작 사양:
//   - 151×92 플레이스홀더를 3열×3행으로 배치. 1행은 처음부터 표시. 스크롤 양: 0.35~1.1초에 0→110px, 1.7~2.5초에 110→225px, 3.5~4.5초에 225→0(easeInOut 3차)
//   - 플레이스홀더 위쪽 끝이 뷰포트 아래쪽 끝(168px)보다 위로 들어온 시점에 로딩 시작. 열마다 왼쪽 열보다 0.12초씩 늦춘다
//   - 로딩 중 0.45초 동안 플레이스홀더 가운데에서 스피너(반지름 11, 0.8초에 1회전)가 돈다
//   - 그다음 0.35초 동안 이미지 불투명도 0→1(easeOut 3차). 로딩 전 플레이스홀더에는 이미지 아이콘을 그린다
//   - 화면 아래로 벗어난 플레이스홀더는 보이지 않는 곳에서 로딩 전 상태로 돌아간다
// 쓰임새: 이미지가 많은 페이지에서 첫 화면 표시를 빠르게 하고 싶을 때.
// 태그: lazy-loading, lazy-load, images, placeholder, performance, spinner, gallery
// 크기: 539x208 px · 1주기: 4.8 s
// Needs: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 움직이는 견본: https://showheyohtaki.github.io/motion-prompt/#090
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 18
// 18 遅延読み込み（画面に入った画像だけ、その時点で読み込まれて表示される）
CELLS[17] = (k, w, h, u) => {
  frame(k);
  const SY = t => steps([[0.35, 0], [1.1, 110], [1.7, 110], [2.5, 225], [3.5, 225], [4.5, 0]], t), y = SY(u);
  const COLX = [15, 180, 345], IW = 151, IH = 92, ROWS = [12, 168, 276], WAIT = 0.45, FADE = 0.35;
  view(k, () => {
    k.save(); k.translate(0, -y);
    txt(k, 'ギャラリー', 15, 140, `800 18px ${JP}`, C.ink); bars(k, 120, 128, 200, 1);
    ROWS.forEach((top, r) => COLX.forEach((x, c) => {
      const t0 = r === 0 ? -1e9 : enteredAt(SY, top, VH, u), age = t0 === null ? -1 : u - t0 - c * 0.12;
      k.fillStyle = C.skel; rr(k, x, top, IW, IH, 6); k.fill();
      if (age < WAIT + FADE) {   // まだの画像: 灰色の枠と画像の印
        k.save(); k.translate(x + IW / 2, top + IH / 2); k.strokeStyle = '#B9C1CF'; k.lineWidth = 2; k.lineJoin = 'round';
        if (age >= 0 && age < WAIT) spinner(k, 0, 0, 11, u, C.blue);
        else { rr(k, -15, -11, 30, 22, 3); k.stroke(); k.beginPath(); k.moveTo(-11, 7); k.lineTo(-3, -1); k.lineTo(3, 5); k.lineTo(7, 1); k.lineTo(11, 6); k.stroke(); }
        k.restore();
      }
      if (age >= WAIT) { k.save(); k.globalAlpha = eOut(P(age, WAIT, WAIT + FADE)); pic(k, x, top, IW, IH, (r + c) % 3, 6); k.restore(); }
    }));
    k.restore();
    sbar(k, y / 225, 0.45);
  });
};
