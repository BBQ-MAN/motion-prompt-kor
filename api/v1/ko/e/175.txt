// 고정 샷 / Locked-off shot
// 팩: camera2 (카메라 워크와 화면 전환 2) · element 07 · 카테고리: 카메라 무빙
// 무엇이 일어나는가: 삼각대에 고정한 카메라 앞을 인물이 가로지르고 뒤로 차가 지나간다. 카메라는 1픽셀도 움직이지 않는다.
// 동작 사양:
//   - 카메라: 완전히 고정(높이 1.4m, 정면 방향, 초점 거리 330px). 피사체만 움직인다
//   - 피사체: 인물이 6.5m 앞을 왼쪽에서 오른쪽으로 일정한 속도로 가로지른다(4.7초에 x = -6.2m → +6.2m). 다리와 팔은 sin(u×7)×0.42rad로 흔들린다
//   - 회색 차가 15m 거리를 오른쪽에서 왼쪽으로 지나간다(4.7초에 x = +16m → -16m)
//   - 둘 다 화면 밖에서 시작해 화면 밖에서 끝나므로 4.7초마다 이음매 없이 반복된다
//   - 왼쪽 아래에 회색 삼각대 마크, 오른쪽 위에 "LOCKED" 표시(#040 프리즈 프레임은 영상의 시간을 멈춘다. 여기서는 시간이 흐르고 카메라만 멈춰 있다)
// 쓰임새: 대비로 피사체를 돋보이게 할 때, 정점 관찰, 타임랩스나 합성의 바탕 화면이 필요할 때.
// 태그: locked-off, locked off, static shot, fixed camera, tripod, static, still camera
// 크기: 539x208 px · 1주기: 9.4 s
// Needs: kit camera2 -> https://bbq-man.github.io/motion-prompt-kor/api/v1/kit/camera2.js
// 움직이는 견본: https://bbq-man.github.io/motion-prompt-kor/#175
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 07
// 07 ロックオフ: 三脚で固定したカメラの前を人物が横切り、奥を車が通る。カメラは1pxも動かない（#040 フリーズフレームは映像が止まる）
CELLS[6] = shot((k, w, h, u) => {
  const q = u / CYC;
  world(k, w, h, cam({ y: 1.4 }), {
    person: { x: lerp(-6.2, 6.2, q), z: 6.5, walk: u * 7 },
    extra: [[lerp(16, -16, q), 0, 15, (kk) => { kk.fillStyle = C.g4; rr(kk, -1.9, 0.35, 3.8, 0.75, 0.25); kk.fill(); rr(kk, -1.1, 0.95, 2.1, 0.62, 0.28); kk.fill(); kk.fillStyle = C.ink; [-1.15, 1.15].forEach(x => { kk.beginPath(); kk.arc(x, 0.34, 0.34, 0, 7); kk.fill(); }); }]],
    after: (kk) => {   // 三脚の印（左下）
      kk.strokeStyle = C.g4; kk.lineWidth = 2.2; kk.lineCap = 'round'; const x = -w / 2 + 38, y = h / 2 - 20;
      kk.beginPath(); kk.moveTo(x, y - 30); kk.lineTo(x - 12, y); kk.moveTo(x, y - 30); kk.lineTo(x + 12, y); kk.moveTo(x, y - 30); kk.lineTo(x, y - 2); kk.stroke();
      kk.fillStyle = C.g4; rr(kk, x - 9, y - 42, 18, 12, 3); kk.fill();
    }
  });
  return 'LOCKED';
});
