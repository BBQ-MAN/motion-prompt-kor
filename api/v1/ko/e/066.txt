// 팩: ui (UI 컴포넌트 이름) · element 18 · 카테고리: 내비게이션·상태 표시
// 무엇이 일어나는가: 브레드크럼의 "강좌"를 누르면 한 단계 위 목록 페이지로 가고, 목록의 카드를 누르면 다시 상세 페이지로 돌아온다.
// 동작 사양:
//   - 4.5초 주기, 카드 542×191. 위쪽에 "홈 › 강좌 › 웹 디자인"(화면은 일본어, 17px): 링크는 파랑, 현재 페이지는 검정 굵게. y 46에 구분선.
//   - 커서가 "강좌"에 올라가면(0.9–1.2초 밑줄) 1.0초에 클릭한다.
//   - 상위 이동 정도: 1.0–1.25초에 easeInOutCubic으로 0→1. "› 웹 디자인"이 사라지고 "강좌"가 현재 페이지(검정 굵게)가 된다.
//   - 내용은 상세 페이지(제목과 본문)에서 카드 3장(158×118, 풍경 그림) 목록으로 크로스페이드된다.
//   - 되돌아가기: 커서가 첫 카드에 올라가(2.45–2.8초 파란 테두리) 2.7초에 클릭하고, 2.7–2.95초에 상세 페이지가 돌아온다.
// 쓰임새: 계층이 깊은 사이트에서 현재 위치와 상위로 돌아가는 길을 보여 줄 때.
// 태그: breadcrumbs, breadcrumb, trail, hierarchy, navigation, path
// 크기: 542x191 px · 1주기: 4.5 s
// Needs: kit ui -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/ui.js
// 움직이는 견본: https://showheyohtaki.github.io/motion-prompt/#066
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 18
// 18 パンくずリスト（Breadcrumbs）
CELLS[17] = ui((w, h, t) => {
  const up = eio(seg(t, 1.0, 1.25)) * (1 - eio(seg(t, 2.7, 2.95))); // 1=「講座」の階層にいる
  let x = 24; const y = 24;
  text('ホーム', x, y, 17, C.blue, 600); x += tw('ホーム', 17, 600) + 10; text('›', x, y, 17, '#aaa', 500); x += 18;
  const kx = x, kw = tw('講座', 17, 700);
  const hoverK = t > .9 && t < 1.2;
  text('講座', x, y, 17, up > .5 ? C.ink : C.blue, up > .5 ? 700 : 600); if (hoverK) { X.fillStyle = C.blue; X.fillRect(x, y + 12, kw, 2); }
  x += kw + 10;
  X.globalAlpha = 1 - up; text('›', x, y, 17, '#aaa', 500); text('Web制作', x + 18, y, 17, C.ink, 700); X.globalAlpha = 1;
  X.fillStyle = C.line; X.fillRect(24, 46, 494, 1.5);
  // 中身：詳細ページ ⇄ 一覧ページ
  if (up < 1) { X.save(); X.globalAlpha = 1 - up; bar(24, 64, 200, C.ink, 13); bar(24, 92, 494); bar(24, 110, 420); fillRR(24, 132, 494, 48, 8, C.block); X.restore(); }
  if (up > 0) { X.save(); X.globalAlpha = up; for (let k = 0; k < 3; k++) { const cx = 24 + k * 168, hov = k === 0 && t > 2.45 && t < 2.8; strokeRR(cx, 62, 158, 118, 8, hov ? C.blue : C.line2, hov ? 2 : 1.5); mountains(cx + 10, 72, 138, 58, k % 2); bar(cx + 10, 142, 90, k === 0 ? C.sb : C.line2, 10); bar(cx + 10, 160, 120, C.line, 7); } X.restore(); }
  pointer([[.3, w + 30, h + 30], [.95, kx + kw / 2, 30], [1.6, kx + kw / 2, 30], [2.4, 104, 120], [3.1, 104, 120], [3.9, w + 30, h + 30]], [1.0, 2.7], t);
});
