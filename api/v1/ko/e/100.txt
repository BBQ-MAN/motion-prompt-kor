// 스플릿 라이팅 / Split lighting
// 팩: photo (사진 표현·빛과 스타일) · element 04 · 카테고리: 조명·시간대
// 무엇이 일어나는가: 빛이 정면 쪽(20°)에서 정측면(90°)으로 돌아가면서 머리의 그림자가 초승달 모양에서 정확히 절반으로 커지고, 얼굴이 콧날을 따라 세로로 명암 둘로 나뉜다.
// 동작 사양:
//   - 장면 1: 지평선(y 150) 앞에 가슴 위까지 보이는 인물(머리 중심 x 270, y 78, 반지름 29px, 얼굴·머리카락 없음). 배경은 한 단계 어두운 장면(하늘 #E7E5E1)으로 바뀐다
//   - 빛: 화면 왼쪽에서. 각도 20° → 90°(오른쪽 아래 배치도, 오른쪽 위 표시 "90° · ↑0°")
//   - 머리: 명암 경계 x = 270 + cos(각도)·√(29² − dy²)보다 오른쪽은 그림자 #33405A, 왼쪽은 빛 #FFC75A. 90°에서는 중심선이 곧은 경계가 된다
//   - 몸통(반폭 56px): 경계 x = 270 + cos(각도)·56보다 오른쪽은 그림자, 왼쪽 가장자리 띠는 빛
//   - 시간: 0.4~1.4초에 효과가 들어오고 4.0초까지 유지된 뒤 4.8초까지 원래대로 돌아간다(1회 5초, 3차 이즈 인 아웃(cubic-bezier(0.65,0,0.35,1) 상당))
// 쓰임새: 강렬하거나 긴장감 있는, 무드 있는 인물 사진을 만들 때.
// 태그: split lighting, side light, portrait lighting, half face, dramatic, chiaroscuro
// 크기: 539x208 px · 1주기: 5 s
// Needs: kit photo -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/photo.js
// 움직이는 견본: https://showheyohtaki.github.io/motion-prompt/#100
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 04
// スプリットライティング: 光が真横（90°）へ回り、顔の右半分が影に沈む
CELLS[3] = cell((k, u) => {
  const a = env(u), az = lerp(0.35, Math.PI / 2, a);
  stagePortrait(k, pal(BASE, DIM, a));
  const lit = mixc(BASE.person, BASE.lit, a), sh = mixc(BASE.person, BASE.shade, a), W = torsoW(PR);
  bust(k, PX, PY, PR, BASE.person, kk => {
    kk.fillStyle = lit; kk.beginPath(); kk.arc(PX, PY, PR, 0, TAU); kk.fill();
    headShade(kk, PX, PY, PR, az, 1, sh);
    torso(kk, PX, PY, PR, PX - W, PX - Math.cos(az) * 0.4 * W - 0.4 * W, lit);
    torso(kk, PX, PY, PR, PX + Math.cos(az) * W, PX + W, sh);
  });
  plan(k, [[az, a]]);
  hud(k, `${Math.round(az * 180 / Math.PI)}° · ↑0°`);
});
