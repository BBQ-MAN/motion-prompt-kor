// 팩: photo (사진 표현·빛과 스타일) · element 08 · 카테고리: 조명·시간대
// 무엇이 일어나는가: 해가 완전히 지면 하늘이 짙은 파랑 두 단으로 바뀌고, 건물 창과 가로등이 앰버로 켜지며, 인물은 어두운 파란 형태가 된다.
// 동작 사양:
//   - 장면 1: 지평선(y 150) 앞에 가슴 위까지 보이는 인물(머리 중심 x 270, y 78, 반지름 29px, 얼굴·머리카락 없음).
//   - 해: 반지름 18px, x 120에서 y 40 → y 190으로 진다(효과량이 70%에 이를 때까지). 40~80% 구간에서 사라진다
//   - 색: 하늘 위쪽 #2C3B5E, 지평선 근처 #4A6191, 땅 #283452, 건물 #34446B, 인물 #1C2439
//   - 불빛: 창 3개 중 1개와 가로등 전구가 #FFC75A. 전구 주위에 반지름 13px·8px의 평평한 빛 고리(#FFE3A3, 35%·60%)
//   - 오른쪽 위 시각 12:00 → 18:35
//   - 시간: 0.4~1.4초에 효과가 들어오고 4.0초까지 유지된 뒤 4.8초까지 원래대로 돌아간다(1회 5초, 3차 이즈 인 아웃(cubic-bezier(0.65,0,0.35,1) 상당))
// 쓰임새: 완전히 어두워지기 전의 파란 하늘과 도시 불빛을 함께 담는 도시·야경 장면에.
// 태그: blue hour, twilight, dusk, night city, cityscape, deep blue sky
// 크기: 539x208 px · 1주기: 5 s
// Needs: kit photo -> https://bbq-man.github.io/motion-prompt-kor/api/v1/kit/photo.js
// 움직이는 견본: https://bbq-man.github.io/motion-prompt-kor/#104
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 08
// ブルーアワー: 日が沈んだ後、空が深い青になり、窓と街灯の暖かい光が灯る
CELLS[7] = cell((k, u) => {
  const a = env(u);
  const p = pal(BASE, NIGHT, a);
  stagePortrait(k, p, {
    sun: [120, lerp(40, 190, P(a, 0, 0.7)), 18, 1 - P(a, 0.4, 0.8)],
    after: D => { const k2 = D.k; k2.save(); k2.globalAlpha = 0.35 * a; D.circ(413, HZ - 53, 13, BASE.glow); k2.globalAlpha = 0.6 * a; D.circ(413, HZ - 53, 8, BASE.glow); k2.restore(); D.circ(413, HZ - 53, 4, p.bulb); }
  });
  bust(k, PX, PY, PR, p.person);
  hud(k, clock(lerp(720, 1115, a)));
});
