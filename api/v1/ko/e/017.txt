// 팩: motion (모션 그래픽 동작) · element 17 · 카테고리: 전환·강조 연출
// 무엇이 일어나는가: 원이나 별 모양의 창이 커지고, 그 안에 다음 장면이 보인다.
// 동작 사양:
//   - 낮 장면 → 밤 장면 → 낮 장면, 각 2.35초. 각 구간의 0.45–1.6초(1.15초)에 전환, 3차 easeInOut.
//   - 현재 장면 위에 도형으로 클리핑한 다음 장면을 겹친다. 도형이 커져 화면 전체를 덮으면 완료.
//   - 낮→밤: 해의 중심(112, 104)에서 원, 반지름 0→470.
//   - 밤→낮: 달의 중심(427, 104)에서 오각별, 바깥 반지름 0→1150. 커지면서 1.2rad(약 69°) 회전.
//   - 커지는 동안 도형 가장자리에 흰 외곽선(굵기 7).
// 쓰임새: 화면 속 사물(해, 로고, 버튼)을 기점으로 다음 장면으로 넘어갈 때.
// 태그: mask, transition, iris, wipe, shape-wipe, reveal, scene-change
// 크기: 539x208 px · 1주기: 4.7 s
// Needs: kit motion -> https://bbq-man.github.io/motion-prompt-kor/api/v1/kit/motion.js
// 움직이는 견본: https://bbq-man.github.io/motion-prompt-kor/#017
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 17
// 17 マスクトランジション（丸や星の形の「窓」が広がり、その中に次の画面が見える）
CELLS[16] = abCell((k, w, h, u, a, b) => {
  const p = eIO(P(u, TA - 0.05, TB + 0.1));
  SCR[a](k, w, h);
  if (p > 0) {
    const c = a === 0 ? SUN : MOON, shape = () => { if (a === 0) { k.beginPath(); k.arc(c[0], c[1], lerp(0, 470, p), 0, 7); } else star(k, c[0], c[1], lerp(0, 1150, p), p * 1.2, 0.5); };
    k.save(); shape(); k.clip(); SCR[b](k, w, h); k.restore();
    if (p < 1) { k.strokeStyle = '#fff'; k.lineWidth = 7; k.lineJoin = 'round'; shape(); k.stroke(); }
  }
});
