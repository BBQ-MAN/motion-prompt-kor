// 스크롤 다운 인디케이터 / Scroll-down cue
// 팩: lp2 (랜딩 페이지·웹사이트 연출 2탄) · element 23 · 카테고리: 시선을 끄는 연출
// 무엇이 일어나는가: 히어로 아래에서 "SCROLL" 문구, 마우스 윤곽 안의 점, 양옆 화살표가 아래 방향 움직임을 계속 반복한다.
// 동작 사양:
//   - 511 × 168 뷰포트. 히어로 아래 가운데에 "SCROLL"(11 px, 자간 3 px)과 마우스 윤곽(24 × 38, radius 12, 선 두께 2.2). 1비트 1.57초(4.7초에 3회, 팩 주기 9.4초에 6회).
//   - 안쪽 주황 점(5 × 9)이 비트의 처음 70% 동안 14 px 내려가고(quadratic ease-in-out), 45%에서 75% 사이에 사라졌다가 다음 비트에 위에서 다시 나타난다.
//   - 좌우로 40 px 떨어진 곳에 아래 방향 셰브론(선 두께 2.5)이 한쪽에 2개씩, 반 비트 차이로 24 px 아래로 흘러간다. opacity는 사인 곡선으로 0 → 0.8 → 0.
// 쓰임새: 히어로가 화면을 꽉 채워서 아래에 내용이 더 있다는 힌트가 필요할 때.
// 태그: scroll-cue, scroll-indicator, scroll-down, hero, arrow, loop
// 크기: 539x208 px · 1주기: 9.4 s
// Needs: kit lp2 -> https://bbq-man.github.io/motion-prompt-kor/api/v1/kit/lp2.js
// 움직이는 견본: https://bbq-man.github.io/motion-prompt-kor/#167
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 23
// 23 スクロールダウンの合図（ファーストビューの下で「SCROLL」とマウスの形の中の点が、下へ促す動きをくり返す）
CELLS[22] = (k, w, h, u) => {
  u = snapU(u);
  frame(k);
  const T = CYC / 3, f = mod(u, T) / T;
  view(k, () => {
    k.fillStyle = '#FFF4EA'; k.fillRect(0, 0, VW, VH);
    txt(k, 'ようこそ、', 40, 40, `800 22px ${JP}`, C.ink); txt(k, '山の上の宿へ。', 40, 70, `800 22px ${JP}`, C.ink);
    bars(k, 40, 84, 170, 2, '#EAD9CB');
    pic(k, 286, 10, 200, 92, 1, 10);
    // 合図
    const mx = VW / 2;
    txt(k, 'SCROLL', mx, 116, `800 11px ${EN}`, C.ink, 'center', 'middle', '3px');
    k.strokeStyle = C.ink; k.lineWidth = 2.2; rr(k, mx - 12, 126, 24, 38, 12); k.stroke();
    const dy = eIOs(clamp(f / 0.7)) * 14, da = 1 - clamp((f - 0.45) / 0.3);
    k.fillStyle = `rgba(255,90,44,${da})`; rr(k, mx - 2.5, 133 + dy, 5, 9, 2.5); k.fill();
    // 両わきの矢印も下へ流れる
    [mx - 40, mx + 40].forEach(x => { for (let j = 0; j < 2; j++) { const ff = mod(f + j * 0.5, 1), yy = 128 + ff * 24, al = Math.sin(Math.PI * ff); k.strokeStyle = `rgba(28,35,51,${0.8 * al})`; k.lineWidth = 2.5; k.lineCap = 'round'; k.lineJoin = 'round'; k.beginPath(); k.moveTo(x - 8, yy); k.lineTo(x, yy + 8); k.lineTo(x + 8, yy); k.stroke(); } });
  });
};
