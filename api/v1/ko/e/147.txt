// 스태킹 카드 / Stacking cards
// 팩: lp2 (랜딩 페이지·웹사이트 연출 2탄) · element 03 · 카테고리: 스크롤 연출
// 무엇이 일어나는가: 스크롤하면 다음 카드가 아래에서 올라와 이전 카드 위에 살짝 어긋나게 겹치고, 아래 깔린 카드는 조금 작아지며 어두워진다.
// 동작 사양:
//   - 511 × 168 뷰포트. 스크롤 단계 s: 0.3–1.1초에 0→1, 1.6–2.4초에 1→2, 3.2–4.4초에 2→0(cubic ease-in-out). 1회 4.7초(팩 주기 9.4초에 2회).
//   - 카드 3장(371 × 124, radius 12, 가운데 정렬), 페이지상 간격 170 px. s가 1 늘 때마다 170 px 위로 이동한다.
//   - 카드는 위에서 14, 26, 38 px 위치에 멈춘다(position: sticky, top 오프셋이 카드마다 12 px씩 커짐). 그래서 새 카드가 이전 카드를 덮는다.
//   - 위에 카드가 한 장 쌓일 때마다 아래 카드는 윗변 중앙을 기준으로 4.5%(두 장이면 9%) 줄어들고, 네이비 레이어(최대 16%)로 어두워진다.
//   - 카드 그림자: blur 14 px, 위로 2 px. 스크롤바 썸은 스크롤을 따라간다.
// 쓰임새: 단계, 기능, 요금제를 순서대로 한 장씩 보여 줄 때.
// 태그: stacking-cards, sticky, scroll, cards, position-sticky, steps
// 크기: 539x208 px · 1주기: 9.4 s
// Needs: kit lp2 -> https://bbq-man.github.io/motion-prompt-kor/api/v1/kit/lp2.js
// 움직이는 견본: https://bbq-man.github.io/motion-prompt-kor/#147
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 03
// 03 スタッキングカード（スクロールすると、次のカードが下から来て前のカードに少しずらして重なる）
CELLS[2] = (k, w, h, u) => {
  u = snapU(u);
  frame(k);
  const s = steps([[0.3, 0], [1.1, 1], [1.6, 1], [2.4, 2], [3.2, 2], [4.4, 0]], u);
  const CARDS = [[C.blueL, C.ink, C.acc, '01', '企画する', 0], [C.ink, '#fff', C.acc, '02', 'つくる', 2], [C.acc, '#fff', '#fff', '03', '届ける', 1]];
  const X = 70, CWD = 371, CHT = 124, STEP = 170;
  view(k, () => {
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH);
    CARDS.forEach(([bg, fg, nc, no, ttl, pv], i) => {
      const top = Math.max(14 + i * 12, 16 + i * STEP - s * STEP);
      const cov = clamp(s - i), sc = 1 - 0.045 * cov - 0.045 * clamp(s - i - 1);
      k.save(); k.translate(VW / 2, top); k.scale(sc, sc); k.translate(-VW / 2, 0);
      k.save(); k.shadowColor = 'rgba(28,35,51,.22)'; k.shadowBlur = 14; k.shadowOffsetY = -2; k.fillStyle = bg; rr(k, X, 0, CWD, CHT, 12); k.fill(); k.restore();
      txt(k, no, X + 24, 44, `800 26px ${EN}`, nc);
      txt(k, ttl, X + 24, 78, `800 20px ${JP}`, fg);
      bars(k, X + 24, 92, 140, 2, i === 0 ? '#C3CDEA' : 'rgba(255,255,255,.35)');
      pic(k, X + CWD - 160, 14, 144, CHT - 28, pv, 8);
      if (cov > 0) { k.fillStyle = `rgba(28,35,51,${0.16 * cov})`; rr(k, X, 0, CWD, CHT, 12); k.fill(); }
      k.restore();
    });
    sbar(k, s / 2, 0.4);
  });
};
