// 무엇이 일어나는가: 거리가 밤으로 바뀌고, 달리는 자동차 차체는 거의 보이지 않을 만큼 옅어지며, 뒤로 빨간 테일램프 선이 늘어나고 반대 차선에는 흰색과 앰버 빛의 선이 지나간다.
// 동작 사양:
//   - 장면 2: 길모퉁이. 안쪽에 건물 줄·나무·가로등(x 114, 292, 494), 도로는 y 132~162, 인물은 왼쪽 앞(발 위치 x 176, y 200, 키 124px). 색은 밤으로 바뀐다(하늘 #2C3B5E, 창 3개 중 1개가 #FFC75A)
//   - 자동차: 5초마다 x −60 → 600. 차체 불투명도 100% → 10%
//   - 테일램프 선: x −60에서 자동차 위치까지. y 145에 #FF5B4D 2.6px, y 149에 #FF8A6E 1.4px
//   - 반대 차선 선: 화면 폭 전체. y 135 #FFE9B8 2.6px, y 138.5 #FFC75A 1.6px, y 141.5 #FFE9B8 1.2px(효과량에 따라 진해진다)
//   - 오른쪽 위 셔터 속도 1/250 → 8″
//   - 시간: 0.4~1.4초에 효과가 들어오고 4.0초까지 유지된 뒤 4.8초까지 원래대로 돌아간다(1회 5초, 3차 이즈 인 아웃(cubic-bezier(0.65,0,0.35,1) 상당))
// 쓰임새: 차량 흐름이나 물·구름의 매끄러운 움직임을 한 장에 담을 때.
// 태그: long exposure, light trails, slow shutter, night photography, car trails, motion
// 크기: 539x208 px · 1주기: 5 s
// Needs: kit photo -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/photo.js
// 움직이는 견본: https://showheyohtaki.github.io/motion-prompt/#107

// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 11
// 長時間露光: シャッターを開け続けると、夜道の車は消え、ライトだけが光の線になる
CELLS[10] = cell((k, u) => {
  const a = env(u), x = lerp(-60, 600, u / CYC);
  const p = pal(BASE, NIGHT, a);
  stageStreet(k, p, {
    car: false, person: false,
    after: D => {
      const kk = D.k;
      // 車の通った跡: 尾灯（赤）と前照灯（白〜アンバー）の線。車の後ろに伸びる
      kk.save(); kk.globalAlpha = a; kk.lineCap = 'round';
      const x0 = -60, x1 = Math.min(x, CW + 40);
      const ln = (ya, yb, c1, c2, w1, w2) => { kk.strokeStyle = c1; kk.lineWidth = w1; kk.beginPath(); kk.moveTo(x0, ya); kk.lineTo(x1, ya); kk.stroke(); kk.strokeStyle = c2; kk.lineWidth = w2; kk.beginPath(); kk.moveTo(x0, yb); kk.lineTo(x1, yb); kk.stroke(); };
      ln(145, 149, '#FF5B4D', '#FF8A6E', 2.6, 1.4);
      // 反対車線の車たちの跡（画面いっぱい）
      for (const [y, c, lw] of [[135, '#FFE9B8', 2.6], [138.5, BASE.lit, 1.6], [141.5, '#FFE9B8', 1.2]]) { kk.strokeStyle = c; kk.lineWidth = lw; kk.beginPath(); kk.moveTo(-10, y); kk.lineTo(CW + 10, y); kk.stroke(); }
      kk.restore();
    }
  });
  // 車の本体は露光が長いほど薄く写る
  k.save(); k.globalAlpha = 1 - 0.9 * a; const D = pen(k);
  car(D, x, p, { head: BASE.lit, tail: '#FF5B4D' }); k.restore();
  figure(pen(k), FIG[0], FIG[1], FIG[2], p.person);
  hud(k, a > 0.5 ? '8″' : '1/250');
});
