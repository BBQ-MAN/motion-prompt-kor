// 팩: camera2 (카메라 워크와 화면 전환 2) · element 14 · 카테고리: 앵글·구도
// 무엇이 일어나는가: 앞쪽의 꽃부터 뒤쪽의 인물과 건물까지 모든 것이 동시에 선명해진다.
// 동작 사양:
//   - 카메라: 위치, 방향, 초점 거리 모두 고정(높이 1.1m). 블러 양만 변한다
//   - 블러: 전경의 꽃(왼쪽 0.75m, 2m 거리)은 5px → 0px, 배경(오른쪽 0.9m, 9m 거리의 인물, 건물, 산)은 6px → 0px로 함께 선명해진다(#033 랙 포커스는 한쪽을 흐린다)
//   - 타이밍: 0.5초부터 3.4초까지 2.9초. 3차 ease-in-out(cubic-bezier(0.65,0,0.35,1)에 가까움)
//   - 초점 마커: 꽃(반폭 46px)과 인물(반폭 30px) 양쪽에 파란 모서리 괄호. 초점이 맞으면서 불투명도 0 → 1
//   - 복귀: 3.4~4.2초 정지 후 마지막 0.5초 동안 시작 상태로 크로스페이드. 오른쪽 위에 조리개 표시(f/1.8 → f/16)
// 쓰임새: 시청자가 화면 어디를 봐도 되게 할 때, 서로 다른 깊이에서 여러 일이 동시에 벌어지는 것을 보여 줄 때.
// 태그: deep focus, depth of field, everything sharp, f/16, small aperture, pan focus, focus
// 크기: 539x208 px · 1주기: 9.4 s
// Needs: kit camera2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/camera2.js
// 움직이는 견본: https://showheyohtaki.github.io/motion-prompt/#182
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 14
// 14 ディープフォーカス: 手前の花から奥の建物まで、全部に同時にピントが合う（#033 ラックフォーカスは片方ずつ）
CELLS[13] = shot(camCell((k, w, h, p) => {
  world(k, w, h, cam({ y: 1.1 }), {
    person: { x: 0.9, z: 9 }, blurFg: 5 * (1 - p), blurBg: 6 * (1 - p),
    extra: [[-0.75, 0, 2.0, (kk, s) => { kk.fillStyle = C.g3; kk.fillRect(-0.025, 0, 0.05, 0.75); kk.fillStyle = C.blueM; for (let i = 0; i < 6; i++) { kk.beginPath(); kk.ellipse(Math.cos(i) * 0.09, 0.8 + Math.sin(i) * 0.09, 0.08, 0.05, i, 0, 7); kk.fill(); } kk.fillStyle = C.blue; kk.beginPath(); kk.arc(0, 0.8, 0.05, 0, 7); kk.fill(); }, true]],
    after: (kk, cc) => {   // ピントの枠を手前と奥の両方に（合うにつれて濃く）
      kk.globalAlpha = p; kk.strokeStyle = C.blue; kk.lineWidth = 2; kk.beginPath();
      [[proj(cc, -0.75, 0.8, 2.0), 46], [proj(cc, 0.9, 0.9, 9), 30]].forEach(([q, r]) => [[-1, -1], [1, -1], [-1, 1], [1, 1]].forEach(([sx, sy]) => { kk.moveTo(q.x + sx * r, q.y + sy * r - sy * 9); kk.lineTo(q.x + sx * r, q.y + sy * r); kk.lineTo(q.x + sx * r - sx * 9, q.y + sy * r); }));
      kk.stroke(); kk.globalAlpha = 1;
    }
  });
}, p => { const a = lerp(1.8, 16, p); return `f/${a < 10 ? a.toFixed(1) : Math.round(a)}`; }));
