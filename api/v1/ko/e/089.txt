// 무엇이 일어나는가: 남색 화면에서 로고 아래 숫자가 0%에서 100%까지 올라가고, 끝나면 화면이 위로 빠지며 아래 페이지가 부드럽게 나타난다.
// 동작 사양:
//   - 뷰포트를 가득 채운 남색 화면에 로고, 숫자(28px), 막대(180×4)
//   - 0.25~2.1초에 0→100%(easeOut 3차, 정수로 반올림). 막대도 같은 비율로 주황색으로 채워진다
//   - 2.3~2.85초에 화면 전체가 위로 168px 빠진다(easeInOut 3차)
//   - 페이지 제목과 사진은 2.35~2.95초에 불투명도 0→1, 14px 아래에서 올라온다(easeOut 3차)
//   - 4.25~4.75초에 화면이 위에서 다시 내려와 0% 상태로 돌아간다(다음 로딩)
// 쓰임새: 사진이나 영상이 많은 사이트에서 로딩 대기 시간을 브랜드를 보여 주는 순간으로 바꾸고 싶을 때.
// 태그: preloader, loading-screen, loader, percentage, intro, splash, curtain
// 크기: 539x208 px · 1주기: 4.8 s
// Needs: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 움직이는 견본: https://showheyohtaki.github.io/motion-prompt/#089
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 17
// 17 ローディング画面（数字が100%まで進むと、幕が上がってページが現れる）
CELLS[16] = (k, w, h, u) => {
  frame(k);
  const n = Math.round(100 * eOut(P(u, 0.25, 2.1)));
  const cy = u < 4.25 ? -VH * eIO(P(u, 2.3, 2.85)) : -VH * (1 - eIO(P(u, 4.25, 4.75)));
  const show = eOut(P(u, 2.35, 2.95));
  view(k, () => {
    header(k, 0);
    k.save(); k.globalAlpha = show; k.translate(0, 14 * (1 - show));
    txt(k, 'ようこそ。', 34, 82, `800 26px ${JP}`, C.ink); bars(k, 34, 98, 200, 3); btn(k, 34, 136, 96, 24, '始める');
    pic(k, 270, 46, 214, 112, 0, 8); k.restore();
    if (cy > -VH) {
      k.save(); k.translate(0, cy); k.fillStyle = C.ink; k.fillRect(0, 0, VW, VH);
      logo(k, VW / 2 - 40, 56, '#fff');
      txt(k, (u >= 4.25 ? 0 : n) + '%', VW / 2, 98, `800 28px ${EN}`, '#fff', 'center', 'middle');
      k.fillStyle = 'rgba(255,255,255,.15)'; rr(k, VW / 2 - 90, 122, 180, 4, 2); k.fill();
      k.fillStyle = C.acc; rr(k, VW / 2 - 90, 122, 180 * (u >= 4.25 ? 0 : n / 100), 4, 2); k.fill();
      k.restore();
    }
  });
};
