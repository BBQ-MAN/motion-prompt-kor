// 스크롤 텍스트 리빌 / Scroll text reveal
// 팩: lp2 (랜딩 페이지·웹사이트 연출 2탄) · element 02 · 카테고리: 스크롤 연출
// 무엇이 일어나는가: 연한 회색 헤드라인이 스크롤에 따라 왼쪽 위부터 한 글자씩 진해지고, 위로 스크롤하면 끝에서부터 다시 옅어진다.
// 동작 사양:
//   - 511 × 168 뷰포트. 스크롤 진행도 p: 0.3–2.6초에 0→1, 3.2–4.4초에 1→0(cubic ease-in-out). 페이지는 p × 40 px만큼 위로 이동한다. 1회 4.7초(팩 주기 9.4초에 2회).
//   - 3줄 헤드라인, 총 24자(27 px, weight 800, 행간 38 px). 시작 시 모두 연한 회색(#D6DBE4).
//   - n번째 글자(0부터 셈)의 진하기는 clamp(p × 26 − n). 각 글자는 진행도의 1/26 동안 본문 색(#1C2333)에 도달하며, 스크롤과 정확히 연동된다.
//   - 되돌려 스크롤하면 같은 식으로 마지막 글자부터 옅어진다. 본문 막대와 스크롤바 썸은 페이지와 함께 움직인다.
// 쓰임새: 콘셉트나 대표 메시지처럼 천천히 읽혀야 하는 짧은 문장에.
// 태그: scroll, text-reveal, scroll-driven, typography, opacity, storytelling
// 크기: 539x208 px · 1주기: 9.4 s
// Needs: kit lp2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp2.js
// 움직이는 견본: https://showheyohtaki.github.io/motion-prompt/#146
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 02
// 02 スクロールテキストリビール（薄い見出しの文字が、スクロールに合わせて左から1文字ずつ濃くなる）
CELLS[1] = (k, w, h, u) => {
  u = snapU(u);
  frame(k);
  const p = steps([[0.3, 0], [2.6, 1], [3.2, 1], [4.4, 0]], u), y = p * 40;
  const LINES = ['毎日の小さな工夫が、', '暮らしを少しずつ', '変えていく。'], F = `800 27px ${JP}`;
  const total = LINES.join('').length;
  view(k, () => {
    k.save(); k.translate(0, -y);
    txt(k, 'ABOUT US', 40, 34, `800 11px ${EN}`, C.acc, 'left', 'alphabetic', '2px');
    let n = 0;
    LINES.forEach((s, li) => {
      let x = 40; k.font = F;
      for (const ch of s) {
        const f = clamp(p * (total + 2) - n);
        txt(k, ch, x, 72 + li * 38, F, mix('#D6DBE4', C.ink, f));
        k.font = F; x += k.measureText(ch).width; n++;
      }
    });
    bars(k, 40, 168, 300, 3, C.faint, 6, 12, 0.7);
    k.restore();
    sbar(k, p, 0.55);
  });
};
