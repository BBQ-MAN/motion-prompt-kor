// 무엇이 일어나는가: 헤더의 "제품"에 커서를 올리면 전체 너비 패널이 내려와 링크 3열과 이미지가 놓인다. 커서가 벗어나면 닫힌다.
// 동작 사양:
//   - 4.5초 주기, 카드 542×191. 높이 42의 헤더에 "제품, 요금, 지원"(화면은 일본어, 16px, 각각 아래 셰브런).
//   - 커서는 올리기만 한다: 1.0초에 "제품"에 닿는다. 열림 정도 a: 1.05–1.3초에 easeOutCubic으로 0→1. "제품"은 파랑에 밑줄.
//   - 패널: 흰색, 전체 너비, 높이 146. 헤더 아래에서 12px 내려와 자리 잡으며 a로 페이드인. 그림자 blur 18, y 오프셋 8.
//   - 안: 130px 간격의 3열(각각 파란 제목과 링크 3개)과 오른쪽에 104×112 그림. 커서가 내려가며 1.75초부터 2.9초까지 2열의 두 번째 링크가 연한 파랑으로 강조된다.
//   - 닫기: 2.9초부터 커서가 패널을 벗어나고, 3.15–3.4초에 (1 − easeInOutCubic)으로 사라진다.
// 쓰임새: 제품이나 섹션이 많은 사이트에서 전체 구성을 한눈에 보여 줄 때.
// 태그: mega-menu, megamenu, dropdown-navigation, navigation, header-menu, multi-column
// 크기: 542x191 px · 1주기: 4.5 s
// Needs: kit ui -> https://bbq-man.github.io/motion-prompt-kor/api/v1/kit/ui.js
// 움직이는 견본: https://bbq-man.github.io/motion-prompt-kor/#067
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 19
// 19 メガメニュー（Mega menu）
CELLS[18] = ui((w, h, t) => {
  const a = eo(seg(t, 1.05, 1.3)) * (1 - eio(seg(t, 3.15, 3.4)));
  pageLines(24, 62, 494); fillRR(24, 130, 494, 50, 8, C.block);
  X.fillStyle = '#fff'; X.fillRect(0, 0, w, 42); X.fillStyle = C.line; X.fillRect(0, 42, w, 1.5);
  fillRR(24, 14, 70, 16, 4, C.mb);
  const nav = [['製品', 190], ['料金', 262], ['サポート', 334]];
  nav.forEach(([s, x], i) => { const on = i === 0 && a > .3; text(s, x, 22, 16, on ? C.blue : '#555', on ? 700 : 600); if (on) fillRR(x, 36, tw(s, 16, 700), 3, 1.5, C.blue); chevron(x + tw(s, 16, 700) + 10, 22, 3.5, 0, on ? C.blue : '#999', 1.8); });
  if (a > 0) {
    X.save(); X.globalAlpha = a; X.beginPath(); X.rect(0, 43, w, h); X.clip(); const py = 43 - 12 * (1 - a);
    shadowed(() => { X.fillStyle = '#fff'; X.fillRect(0, py, w, 146); }, 18, 8, .16);
    for (let c = 0; c < 3; c++) { const cx = 24 + c * 130; bar(cx, py + 16, 80, C.blue, 10); for (let k = 0; k < 3; k++) { const hov = c === 1 && k === 1 && t > 1.75 && t < 2.9; if (hov) fillRR(cx - 6, py + 34 + k * 26, 112, 22, 5, C.lb); bar(cx, py + 40 + k * 26, [90, 70, 100][(k + c) % 3], hov ? C.sb : C.line2, 9); } }
    mountains(414, py + 14, 104, 112);
    X.restore();
  }
  // 乗せるだけで開く（クリックしない）
  pointer([[.3, w + 30, h + 30], [1.0, 205, 26], [1.4, 205, 26], [1.8, 196, 98], [2.9, 196, 98], [3.2, 300, 200], [3.9, w + 30, h + 30]], [], t);
});
