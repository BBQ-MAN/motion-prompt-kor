// 팩: photo (사진 표현·빛과 스타일) · element 03 · 카테고리: 조명·시간대
// 무엇이 일어나는가: 노출을 밝은 저녁 하늘에 맞춰 가면 인물·건물·가로등·나무가 디테일 없는 검은 형태가 되고 하늘만 밝게 남는다.
// 동작 사양:
//   - 장면 1: 지평선(y 150) 앞에 가슴 위까지 보이는 인물(머리 중심 x 270, y 78, 반지름 29px, 얼굴·머리카락 없음).
//   - 하늘: #FAF9F7 → 위쪽 #FFEDC4, 지평선 근처 띠(y 92~150) #FFCF73
//   - 하늘을 뺀 모든 형태는 #2A2D36, 먼 언덕은 #4A4546, 인물은 #22252D
//   - 해: 반지름 20px, x 404에서 y 60 → y 128로 지며 언덕에 반쯤 가려진다
//   - 오른쪽 위 표시: ±0.0 EV → −2.0 EV
//   - 시간: 0.4~1.4초에 효과가 들어오고 4.0초까지 유지된 뒤 4.8초까지 원래대로 돌아간다(1회 5초, 3차 이즈 인 아웃(cubic-bezier(0.65,0,0.35,1) 상당))
// 쓰임새: 노을이나 창문 빛을 배경으로 형태와 포즈만으로 인상을 남기고 싶을 때.
// 태그: silhouette, backlit, sunset, contre-jour, black shape, underexposed
// 크기: 539x208 px · 1주기: 5 s
// Needs: kit photo -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/photo.js
// 움직이는 견본: https://showheyohtaki.github.io/motion-prompt/#099
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 03
// シルエット: 露出を明るい空に合わせると、人物も建物も木も黒い形だけになる
CELLS[2] = cell((k, u) => {
  const a = env(u), INK = '#2A2D36';
  const SILH = mapPal(BASE, () => INK);
  Object.assign(SILH, { sky: '#FFEDC4', sky2: '#FFCF73', far: '#4A4546', sun: BASE.sun, glow: BASE.glow });
  stagePortrait(k, pal(BASE, SILH, a), { sun: [404, lerp(60, 128, a), 20, a] });
  bust(k, PX, PY, PR, mixc(BASE.person, '#22252D', a));
  hud(k, `${(-2 * a).toFixed(1).replace('-0.0', '±0.0')} EV`);
});
