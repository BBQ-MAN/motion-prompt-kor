// 그라데이션 애니메이션 / Animated gradient
// 팩: lp (랜딩 페이지·웹사이트 연출) · element 23 · 카테고리: 로딩·페이지 이동·반복 연출
// 무엇이 일어나는가: 헤드라인 뒤에서 주황, 옅은 주황, 파랑, 옅은 파랑의 흐릿한 색 덩어리가 천천히 돌며 섞이는 모습이 계속 바뀐다.
// 동작 사양:
//   - 바탕색 #FFF1EB 위에 원형 그라데이션 4개(중심 색 → 바깥에서 투명, 각 불투명도 0.85)
//   - 주황색(반지름 230), 옅은 주황 #FFB08F(210), 파란색(240), 옅은 파랑 #9DB6FB(200)
//   - 각 중심은 타원 위를 4.8초에 한 바퀴 돈다(가로 반지름 70, 60, 80, 50, 세로 반지름 30, 40, 30, 35, 시작 각도 0, 2.1, 4.2, 1.0rad)
//   - 위에 흰색 헤드라인(34px)과 버튼. 그라데이션만 움직인다
// 쓰임새: 사진 없이 히어로 섹션이나 제목 배경에 움직임과 색감을 더하고 싶을 때.
// 태그: gradient, animated-gradient, mesh-gradient, background, hero, ambient, loop
// 크기: 539x208 px · 1주기: 4.8 s
// Needs: kit lp -> https://bbq-man.github.io/motion-prompt-kor/api/v1/kit/lp.js
// 움직이는 견본: https://bbq-man.github.io/motion-prompt-kor/#095
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 23
// 23 グラデーションアニメーション（背景のグラデーションの色が、ゆっくり流れるように動き続ける）
CELLS[22] = (k, w, h, u) => {
  frame(k);
  const a = 2 * Math.PI * u / CYC;                        // 4.8秒で1周
  const BLOBS = [[C.acc, 150, 60, 230, 70, 30, 0], ['#FFB08F', 380, 40, 210, 60, 40, 2.1], [C.blue, 330, 150, 240, 80, 30, 4.2], ['#9DB6FB', 90, 170, 200, 50, 35, 1.0]];
  view(k, () => {
    k.fillStyle = '#FFF1EB'; k.fillRect(0, 0, VW, VH);
    for (const [col, x0, y0, r, ax, ay, ph] of BLOBS) {
      const x = x0 + ax * Math.cos(a + ph), y = y0 + ay * Math.sin(a + ph);
      const g = k.createRadialGradient(x, y, 0, x, y, r); g.addColorStop(0, col); g.addColorStop(1, 'rgba(255,255,255,0)');
      k.globalAlpha = 0.85; k.fillStyle = g; k.fillRect(0, 0, VW, VH); k.globalAlpha = 1;
    }
    txt(k, 'Hello, Future.', 40, 82, `800 34px ${EN}`, '#fff', 'left', 'alphabetic', '1px');
    bars(k, 40, 100, 200, 2, 'rgba(255,255,255,.8)', 6, 12, 0.6);
    btn(k, 40, 128, 104, 26, 'はじめる', '#fff', C.ink);
  });
};
