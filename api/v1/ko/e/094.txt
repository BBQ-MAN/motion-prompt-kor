// 무엇이 일어나는가: 남색 띠 안에서 큼직한 영문 서비스 이름과 주황색 점이 끊김 없이 왼쪽으로 계속 흐른다.
// 동작 사양:
//   - 띠: 높이 58, 남색, 뷰포트 폭 전체. 글자: 30px, 굵게, 자간 1px, 흰색
//   - 4단어(WEB DESIGN / BRANDING / MOTION / UI/UX)를 한 세트로 하고, 단어 사이는 64px, 그 가운데에 주황색 점(반지름 6)
//   - 한 주기(4.8초)에 한 세트의 폭(이 환경에서 약 881px)만큼 등속으로 왼쪽으로 이동. 세트 폭을 재서 그 폭으로 위치를 되감으므로 이음매가 보이지 않는다
//   - 뷰포트 폭이 채워질 때까지 같은 세트를 반복해 늘어놓는다
// 쓰임새: 서비스 이름, 고객사 로고, 캐치프레이즈를 활기차게 보여 주고 싶을 때.
// 태그: marquee, ticker, infinite-loop, scrolling-text, logo-strip, band, loop
// 크기: 539x208 px · 1주기: 4.8 s
// Needs: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 움직이는 견본: https://showheyohtaki.github.io/motion-prompt/#094
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 22
// 22 マーキー（文字の帯が、途切れずに横へ流れ続ける）
CELLS[21] = (k, w, h, u) => {
  frame(k);
  const WORDS = ['WEB DESIGN', 'BRANDING', 'MOTION', 'UI/UX'], F = `800 30px ${EN}`, GAPW = 64;
  view(k, () => {
    bars(k, 34, 20, 180, 1, C.ink, 10); bars(k, 34, 38, 300, 1);
    k.fillStyle = C.ink; k.fillRect(0, 58, VW, 58);
    k.font = F; k.letterSpacing = '1px'; const ws = WORDS.map(s => k.measureText(s).width); k.letterSpacing = '0px';
    const unit = ws.reduce((a, b) => a + b + GAPW, 0);
    let x = -mod(u / CYC, 1) * unit;                 // 1周期で1組ぶん左へ
    for (let rep = 0; x < VW; rep++) WORDS.forEach((s, i) => {
      txt(k, s, x, 88, F, '#fff', 'left', 'middle', '1px');
      k.fillStyle = C.acc; k.beginPath(); k.arc(x + ws[i] + GAPW / 2, 88, 6, 0, 7); k.fill();
      x += ws[i] + GAPW;
    });
    for (let i = 0; i < 3; i++) { k.fillStyle = C.soft; rr(k, 34 + i * 152, 130, 136, 30, 6); k.fill(); bars(k, 46 + i * 152, 142, 90, 1, C.faint); }
  });
};
