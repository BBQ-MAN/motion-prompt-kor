// 팩: camera (카메라 무빙과 화면 전환) · element 06 · 카테고리: 카메라 움직임
// 무엇이 일어나는가: 피사체를 중심으로 카메라가 원을 그리며 돈다. 인물은 중앙에 머물고, 뒤쪽 나무와 건물이 옆으로 크게 흐른다.
// 동작 사양:
//   - 카메라: 인물(정면 9m 앞)을 중심으로 한 반지름 6m 원 위를 -0.95rad → +0.95rad(-54° → +54°, 총 약 109°) 이동
//   - 방향: 항상 중심을 향한다(요 = -각도). 높이 1.7m, 피치는 -0.07rad(약 4° 아래)로 고정
//   - 시간: 0.5초부터 3.6초까지 3.1초. 3차 ease-in-out(cubic-bezier(0.65,0,0.35,1) 상당)
//   - 피사체: 인물은 움직이지 않는다. 얼굴의 흰 점이 각도에 따라 옆으로 이동한다. 발밑에 반지름 1.3m의 점선 원
//   - 배경: 반지름 10m 원 위에 나무 12그루를 같은 간격으로 배치. 카메라가 돌면 인물 뒤를 가로지른다
//   - 복귀: 3.6~4.2초는 정지, 마지막 0.5초 동안 시작 상태로 크로스페이드. 오른쪽 위에 각도 표시(-54° → +54°)
// 쓰임새: 인물이나 제품을 입체적으로 보여 주거나, 등장 장면이나 결정적인 순간을 강조할 때.
// 태그: orbit, arc shot, circle, 360, rotate around, revolve, camera move
// 크기: 539x208 px · 1주기: 9.4 s
// Needs: kit camera -> https://bbq-man.github.io/motion-prompt-kor/api/v1/kit/camera.js
// 움직이는 견본: https://bbq-man.github.io/motion-prompt-kor/#030
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 06
// 06 オービット: 被写体を中心に円を描いて回り込む
CELLS[5] = shot(camCell((k, w, h, p) => {
  const a = lerp(-0.95, 0.95, p), R = 6, P0 = [0, 0, 9];
  const c = cam({ x: P0[0] + Math.sin(a) * R, z: P0[2] - Math.cos(a) * R, yaw: -a, y: 1.7, pitch: -0.07 });
  world(k, w, h, c, {
    person: { z: 9, face: a }, ring: [0, 9, 10], buildings: true, trees: false,
    after: (kk, cc) => { kk.strokeStyle = C.blueM; kk.lineWidth = 2; kk.setLineDash([5, 5]); kk.beginPath(); for (let i = 0; i <= 48; i++) { const b = i / 48 * Math.PI * 2; const q = proj(cc, Math.sin(b) * 1.3, 0, 9 + Math.cos(b) * 1.3); i ? kk.lineTo(q.x, q.y) : kk.moveTo(q.x, q.y); } kk.stroke(); kk.setLineDash([]); }
  });
}, p => `${Math.round(lerp(-0.95, 0.95, p) * 57.3)}°`.replace(/^(\d)/, '+$1'), 0.5, 3.6));
