// POV 샷 / POV shot
// 팩: camera2 (카메라 워크와 화면 전환 2) · element 11 · 카테고리: 앵글·구도
// 무엇이 일어나는가: 캐릭터의 눈으로 본 화면. 걸음의 흔들림과 화면 아래의 손이 1인칭임을 알려 주고, 앞의 인물이 이쪽을 향해 손을 흔든다.
// 동작 사양:
//   - 카메라: 캐릭터의 눈높이 1.6m에서 앞으로 0m → 3.2m 걷는다. 0.3초부터 4.2초까지 3.9초, 3차 ease-in-out(cubic-bezier(0.65,0,0.35,1)에 가까움)
//   - 걸음 흔들림: 높이 ±3cm(4.7초에 6주기), 롤 ±0.008rad(3주기). 이동하는 동안에만 더한다
//   - 손: 아래쪽 양 모서리(중심에서 ±178px)에 둥근 파란 손, 오른손에는 흰 휴대폰. 몸과 함께 움직이므로 손은 고정되고 세상이 흔들린다
//   - 7.5m 앞의 인물은 1.6초부터 3.4초까지만 두 팔을 들어 손을 흔든다(이쪽을 보며). 전경에 말뚝 두 개
//   - 복귀: 마지막 0.5초 동안 시작 상태로 크로스페이드(1회 4.7초). 오른쪽 위에 "POV" 표시
// 쓰임새: 시청자를 캐릭터의 입장에 세울 때, 게임이나 VR 같은 느낌을 낼 때.
// 태그: pov, point of view, first person, subjective, eyes, hands, angle
// 크기: 539x208 px · 1주기: 9.4 s
// Needs: kit camera2 -> https://bbq-man.github.io/motion-prompt-kor/api/v1/kit/camera2.js
// 움직이는 견본: https://bbq-man.github.io/motion-prompt-kor/#179
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 11
// 11 POVショット: 登場人物の目で見た画。歩く上下動と画面下の両手で一人称だと分かる。前の人物が振り向いて手を振る
CELLS[10] = shot(camCell((k, w, h, p, u) => {
  const z = lerp(0, 3.2, p), T = u * 2 * Math.PI / CYC;
  const bob = Math.sin(T * 6) * 0.03 * (p > 0 && p < 1 ? 1 : 0);
  world(k, w, h, cam({ z, y: 1.6 + bob, roll: Math.sin(T * 3) * 0.008 * (p > 0 && p < 1 ? 1 : 0) }), {
    person: { z: 7.5, armUp: u > 1.6 && u < 3.4, walk: 0 }, fg: [[-1.8, 2.6], [2.2, 3.8]],
    after: (kk) => {   // 両手（画面の下の両端から入る。体と一緒に動くので、揺れるのは世界のほう）
      kk.fillStyle = C.blue;
      [-1, 1].forEach(sx => { kk.save(); kk.translate(sx * 178, 64); kk.rotate(sx * -0.5); kk.fillStyle = C.blue; rr(kk, -34, -28, 68, 120, 30); kk.fill(); if (sx > 0) { kk.fillStyle = '#fff'; rr(kk, -17, -62, 34, 54, 6); kk.fill(); kk.strokeStyle = C.blue; kk.lineWidth = 3; rr(kk, -17, -62, 34, 54, 6); kk.stroke(); } kk.restore(); });
    }
  });
}, p => 'POV', 0.3, 4.2));
