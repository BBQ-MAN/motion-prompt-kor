// 버터플라이 라이팅 / Butterfly lighting
// 팩: photo (사진 표현·빛과 스타일) · element 06 · 카테고리: 조명·시간대
// 무엇이 일어나는가: 얼굴 정면 높은 곳에서 빛이 들어오면 얼굴이 좌우 고르게 밝아지고, 턱 아래에만 좌우 대칭의 작은 그림자가 진다.
// 동작 사양:
//   - 장면 1: 지평선(y 150) 앞에 가슴 위까지 보이는 인물(머리 중심 x 270, y 78, 반지름 29px, 얼굴·머리카락 없음). 배경은 한 단계 어두운 장면(하늘 #E7E5E1)으로 바뀐다
//   - 머리: 전체가 빛 #FFC75A(그림자 없음)
//   - 턱 아래 그림자: 타원(중심 y 78+1.06R, 가로 0.72R, 세로 0.38R, R = 29 → 약 21×11px), 색 #33405A
//   - 몸통: 중심에서 ±0.8×56px 안은 빛, 바깥쪽 가는 띠는 원래 색. 배치도에서 조명은 카메라 쪽(0°), 오른쪽 위 표시 "0° · ↑45°"
//   - 시간: 0.4~1.4초에 효과가 들어오고 4.0초까지 유지된 뒤 4.8초까지 원래대로 돌아간다(1회 5초, 3차 이즈 인 아웃(cubic-bezier(0.65,0,0.35,1) 상당))
// 쓰임새: 얼굴을 돋보이게 하는 뷰티·패션 인물 사진에서.
// 태그: butterfly lighting, paramount lighting, beauty lighting, portrait lighting, overhead light, glamour
// 크기: 539x208 px · 1주기: 5 s
// Needs: kit photo -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/photo.js
// 움직이는 견본: https://showheyohtaki.github.io/motion-prompt/#102
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 06
// バタフライライティング: 顔の正面・高い位置からの光。顔は左右対称に明るく、あごの下にだけ小さな影
CELLS[5] = cell((k, u) => {
  const a = env(u), R = PR, W = torsoW(R);
  stagePortrait(k, pal(BASE, DIM, a));
  const lit = mixc(BASE.person, BASE.lit, a), sh = mixc(BASE.person, BASE.shade, a);
  bust(k, PX, PY, R, BASE.person, kk => {
    torso(kk, PX, PY, R, PX - 0.8 * W, PX + 0.8 * W, lit);
    kk.fillStyle = sh; kk.beginPath(); kk.ellipse(PX, PY + R * 1.06, R * 0.72, R * 0.38, 0, 0, TAU); kk.fill();    // あごの下の影（左右対称）
    kk.fillStyle = lit; kk.beginPath(); kk.arc(PX, PY, R, 0, TAU); kk.fill();
  });
  plan(k, [[0, a]]);
  hud(k, '0° · ↑45°');
});
