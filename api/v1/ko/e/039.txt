// 팩: camera (카메라 무빙과 화면 전환) · element 15 · 카테고리: 촬영 기법과 시간 연출
// 무엇이 일어나는가: 고정 카메라로 06:00부터 18:00까지를 몇 초로 압축한다. 해, 구름, 사람, 그림자가 빨리 감기처럼 움직인다.
// 동작 사양:
//   - 카메라: 완전히 고정(높이 1.6m, 피치 +0.08rad, 약 5° 위)
//   - 시간: 0.4초부터 4.1초까지 3.7초 동안 12시간이 흐른다. 3차 ease-in-out(cubic-bezier(0.65,0,0.35,1) 상당)
//   - 해: 각도 a = -1.25rad → +1.25rad의 호를 따라 왼쪽에서 오른쪽으로 이동. 위치는 (sin a×120, cos a×60 - 8) m, 깊이 170m
//   - 구름 3개는 오른쪽으로 총 140m 흐르며 80m마다 순환. 작은 사람 7명은 각각 폭 18m를 6번 가로지른다(홀수 번째와 짝수 번째는 반대 방향)
//   - 건물 그림자: 해의 반대쪽으로 드리워지고, 해의 움직임에 따라 방향이 바뀐다(반투명 회색)
//   - 복귀: 마지막 0.5초 동안 시작 상태로 크로스페이드. 오른쪽 위 시계는 06:00 → 18:00을 10분 단위로 표시
// 쓰임새: 긴 시간의 흐름이나 도시, 하늘, 작업의 변화를 짧게 보여 줄 때.
// 태그: time-lapse, timelapse, fast forward, time compression, sun, clouds, day to night
// 크기: 539x208 px · 1주기: 9.4 s
// Needs: kit camera -> https://bbq-man.github.io/motion-prompt-kor/api/v1/kit/camera.js
// 움직이는 견본: https://bbq-man.github.io/motion-prompt-kor/#039
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 15
// 15 タイムラプス: 太陽が動き、雲と人が早送り
CELLS[14] = shot(camCell((k, w, h, p, u) => {
  const a = lerp(-1.25, 1.25, p);
  const clouds = [[-20 + mod(p * 140 + 0, 80) - 10, 18, 90, 2.2], [-50 + mod(p * 140 + 40, 80), 24, 120, 3], [-10 + mod(p * 140 + 20, 80) - 30, 13, 70, 1.6]];
  const ex = [];
  for (let i = 0; i < 7; i++) { const ph = mod(p * 6 + i / 7, 1), dir = i % 2 ? 1 : -1; ex.push([dir * lerp(-9, 9, ph), 0, 10 + i * 1.3, (kk) => { kk.fillStyle = i % 3 ? C.g4 : C.blueM; kk.fillRect(-0.12, 0, 0.24, 0.9); kk.beginPath(); kk.arc(0, 1.05, 0.13, 0, 7); kk.fill(); }]); }
  world(k, w, h, cam({ y: 1.6, pitch: 0.08 }), {
    sun: [Math.sin(a) * 120, Math.cos(a) * 60 - 8, 170], sunCol: C.blueM, clouds, extra: ex,
    after: (kk, cc) => {   // 建物の影
      kk.fillStyle = 'rgba(160,160,165,.22)';
      BLD.slice(0, 6).forEach(([x, bw, bh]) => { const L = bh * 0.9; const q1 = proj(cc, x - bw / 2, 0, 48), q2 = proj(cc, x + bw / 2, 0, 48), q3 = proj(cc, x + bw / 2 - Math.sin(a) * L, 0, 48 - Math.cos(a) * L * 0.4 - 1), q4 = proj(cc, x - bw / 2 - Math.sin(a) * L, 0, 48 - Math.cos(a) * L * 0.4 - 1); kk.beginPath(); kk.moveTo(q1.x, q1.y); kk.lineTo(q2.x, q2.y); kk.lineTo(q3.x, q3.y); kk.lineTo(q4.x, q4.y); kk.fill(); });
    }
  });
}, p => { const m = Math.round(lerp(6 * 60, 18 * 60, p) / 10) * 10; return `${String(Math.floor(m / 60)).padStart(2, '0')}:${String(m % 60).padStart(2, '0')}`; }, 0.4, 4.1));
