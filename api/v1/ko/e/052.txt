// 무엇이 일어나는가: 제목 3줄 중 첫 줄을 누르면 아래로 본문이 펼쳐지며 아래 줄들이 밀려 내려간다. 다시 누르면 접힌다.
// 동작 사양:
//   - 4.5초 주기, 카드 542×191. 제목 3줄(너비 494, 높이 34, 모서리 반경 8, 간격 6), 오른쪽 끝에 아래 방향 셰브런.
//   - 커서가 0.95초에 첫 제목에 닿고 1.0초와 3.3초에 클릭한다.
//   - 열림 정도 ex: 1.0–1.35초에 easeInOutCubic으로 0→1, 3.3–3.62초에 1→0.
//   - 첫 줄 높이가 34에서 92로(+58·ex) 늘어나며 아래 줄을 밀어 내린다. 본문 2줄은 줄 영역 안으로 잘라 불투명도 ex로 나타낸다.
//   - 펼쳐진 동안 셰브런은 180°·ex 회전하고, 테두리는 연한 파랑, 제목은 파랑이 된다.
// 쓰임새: FAQ나 긴 설명처럼 제목만 훑어보고 필요한 내용만 펼쳐 보게 할 때.
// 태그: accordion, collapsible, expand, collapse, disclosure, faq, details
// 크기: 542x191 px · 1주기: 4.5 s
// Needs: kit ui -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/ui.js
// 움직이는 견본: https://showheyohtaki.github.io/motion-prompt/#052
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 04
// 04 アコーディオン（Accordion）
CELLS[3] = ui((w, h, t) => {
  const ex = eio(seg(t, 1.0, 1.35)) * (1 - eio(seg(t, 3.3, 3.62)));
  let y = 12; const H0 = 34, EX = 58;
  for (let i = 0; i < 3; i++) {
    const extra = i === 0 ? EX * ex : 0;
    strokeRR(24, y, 494, H0 + extra, 8, i === 0 && ex > .05 ? C.sb : C.line2, 1.5);
    bar(42, y + 12, [180, 220, 150][i], i === 0 && ex > .05 ? C.blue : C.line2, 10);
    chevron(494, y + 17, 6, i === 0 ? Math.PI * ex : 0, i === 0 && ex > .05 ? C.blue : '#9a9896');
    if (i === 0 && ex > 0) { X.save(); rr(24, y, 494, H0 + extra, 8); X.clip(); X.globalAlpha = ex; bar(42, y + H0 + 6, 420); bar(42, y + H0 + 24, 320); X.restore(); }
    y += H0 + extra + 6;
  }
  pointer([[.3, w + 30, h + 30], [.95, 300, 31], [1.6, 300, 31], [3.0, 300, 31], [3.55, 300, 31], [4.2, w + 30, h + 30]], [1.0, 3.3], t);
});
