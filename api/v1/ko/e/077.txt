// 가로 스크롤 / Horizontal scroll
// 팩: lp (랜딩 페이지·웹사이트 연출) · element 05 · 카테고리: 스크롤 연출
// 무엇이 일어나는가: 아래로 스크롤하면 섹션은 멈춘 채로, 번호가 붙은 작품 카드 4장이 왼쪽으로 흘러간다.
// 동작 사양:
//   - 스크롤 양: 0.3~3.0초에 0→420px, 3.5~4.5초에 420→0(easeInOut 3차)
//   - 제목 'WORKS'는 위로 흘러가고, 섹션은 스크롤 양 40px에서 위쪽 끝에 닿아 멈춘다
//   - 카드: 200×112 4장, 간격 20, 좌우 여백 34. 가로 이동량 = 34×2 + 200×4 + 20×3 − 511 = 417px
//   - 카드 x = 원래 x − p × 417, p = (스크롤 양 − 40) / 380(스크롤에 비례, 이징 없음)
//   - 스크롤바 손잡이는 세로로, 카드는 가로로 움직인다
// 쓰임새: 작품, 사례, 상품을 세로로 긴 페이지 안에서 가로 한 줄로 보여 주고 싶을 때.
// 태그: horizontal-scroll, sideways, scrolltrigger, pin, gallery, portfolio, scroll
// 크기: 539x208 px · 1주기: 4.8 s
// Needs: kit lp -> https://bbq-man.github.io/motion-prompt-kor/api/v1/kit/lp.js
// 움직이는 견본: https://bbq-man.github.io/motion-prompt-kor/#077
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 05
// 05 横スクロール（縦にスクロールすると、カードの列が横へ流れる）
CELLS[4] = (k, w, h, u) => {
  frame(k);
  const y = steps([[0.3, 0], [3.0, 420], [3.5, 420], [4.5, 0]], u);
  const PIN0 = 40, PD = 380, N = 4, PWD = 200, GAP = 20;
  const top = y < PIN0 ? PIN0 - y : 0, p = clamp((y - PIN0) / PD);
  const travel = 34 * 2 + N * PWD + (N - 1) * GAP - VW;
  view(k, () => {
    txt(k, 'WORKS', 34, 30 - y, `800 20px ${EN}`, C.ink, 'left', 'alphabetic', '2px');
    k.save(); k.translate(0, top);
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH + 40);
    for (let i = 0; i < N; i++) {
      const x = 34 + i * (PWD + GAP) - p * travel;
      pic(k, x, 18, PWD, 112, i % 3, 8);
      txt(k, '0' + (i + 1), x + 14, 118, `800 22px ${EN}`, '#fff');
      bars(k, x, 140, 120, 1, '#CBD2DE');
    }
    k.restore();
    sbar(k, y / 420, 0.36);
  });
};
