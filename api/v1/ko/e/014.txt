// 팩: motion (모션 그래픽 동작) · element 14 · 카테고리: 움직임 디테일
// 무엇이 일어나는가: 8자를 그리며 움직이는 공 뒤로 잔상이 꼬리를 남긴다.
// 동작 사양:
//   - 공(코랄, 반지름 25, 가운데 흰 원 반지름 11)이 4.7초에 한 바퀴 8자를 그린다: x = 196·sin(a), y = 62·sin(2a). a는 일정한 속도로 증가.
//   - 잔상 15개를 경로를 따라 0.105rad(약 0.079초)씩 과거 위치에 둔다.
//   - 잔상 반지름은 22→6, 불투명도는 0.9 × (1 − i/15) + 0.05. 오래된 것일수록 작고 옅다.
//   - 잔상 색은 최신부터 코랄 → 노랑 → 파랑.
//   - 가장 오래된 잔상부터 그리고, 공 본체를 마지막에 그린다.
// 쓰임새: 속도나 궤적을 보여 줄 때. 날아가는 물체, 커서, 빛나는 점에.
// 태그: trail, motion-trail, echo, afterimage, ghost, path, speed
// 크기: 539x208 px · 1주기: 4.7 s
// Needs: kit motion -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/motion.js
// 움직이는 견본: https://showheyohtaki.github.io/motion-prompt/#014
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 14
// 14 モーショントレイル（動いた跡に、残像が尾を引く）
CELLS[13] = (k, w, h, u) => {
  stage(k);
  const X = w / 2, Y = h / 2 + 2, A = 196, B = 62, th = u / CYC * Math.PI * 2;
  const pt = a => [X + A * Math.sin(a), Y + B * Math.sin(2 * a)];
  const N = 15;
  for (let i = N; i >= 1; i--) {
    const [x, y] = pt(th - i * 0.105), f = i / N;
    k.globalAlpha = 0.9 * (1 - f) + 0.05; k.fillStyle = f < 0.5 ? mix(C.coral, C.yel, f * 2) : mix(C.yel, C.blue, (f - 0.5) * 2);
    k.beginPath(); k.arc(x, y, lerp(22, 6, f), 0, 7); k.fill();
  }
  k.globalAlpha = 1; const [x, y] = pt(th);
  k.fillStyle = C.coral; k.beginPath(); k.arc(x, y, 25, 0, 7); k.fill();
  k.fillStyle = '#fff'; k.beginPath(); k.arc(x, y, 11, 0, 7); k.fill();
};
