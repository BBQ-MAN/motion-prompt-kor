// 무엇이 일어나는가: 커서가 올라간 카드만 살짝 위로 들리고 아래 그림자가 넓어진다. 커서가 벗어나면 원래대로 돌아온다.
// 동작 사양:
//   - 카드 3장(146×128, 모서리 반경 10). 커서가 왼쪽부터 차례로 약 1초씩 올라간다
//   - 호버하면 0.3초 동안 위로 10px(easeOut 3차). 벗어나면 같은 0.3초로 돌아온다(CSS transition과 같다)
//   - 그림자: 블러 6→22px, 아래 오프셋 2→12px, 불투명도 0.08→0.22를 이동과 함께 바꾼다
//   - 바탕은 옅은 회색, 카드는 흰색(그림자가 보이도록)
// 쓰임새: 카드나 버튼처럼 누를 수 있는 요소를 누를 수 있어 보이게 하고 싶을 때.
// 태그: hover, lift, float, card, shadow, translatey, interaction
// 크기: 539x208 px · 1주기: 4.8 s
// Needs: kit lp -> https://bbq-man.github.io/motion-prompt-kor/api/v1/kit/lp.js
// 움직이는 견본: https://bbq-man.github.io/motion-prompt-kor/#081
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 09
// 09 ホバーで浮き上がる（カーソルが乗ったカードが、影をつけて少し持ち上がる）
CELLS[8] = (k, w, h, u) => {
  frame(k);
  const CX = [26, 192, 358], CY = 18, CWD = 146, CHT = 128, LIFT = 10;
  const cur = t => path([[0, 70, 160], [0.45, 96, 78], [1.4, 110, 86], [1.62, 262, 80], [2.5, 276, 88], [2.72, 428, 80], [3.6, 440, 88], [4.0, 470, 160], [4.8, 70, 160]], t);
  view(k, () => {
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH);
    for (let i = 0; i < 3; i++) {
      const inside = t => { const [x, y] = cur(t); return x > CX[i] && x < CX[i] + CWD && y > CY && y < CY + CHT; };
      const q = trans(inside, u, 0.3), x = CX[i], y = CY - LIFT * q;
      k.save(); k.shadowColor = `rgba(28,35,51,${0.08 + 0.14 * q})`; k.shadowBlur = 6 + 16 * q; k.shadowOffsetY = 2 + 10 * q;
      k.fillStyle = '#fff'; rr(k, x, y, CWD, CHT, 10); k.fill(); k.restore();
      pic(k, x + 8, y + 8, CWD - 16, 66, i, 6);
      txt(k, ['山の宿', '海の宿', '星の宿'][i], x + 12, y + 96, `800 13px ${JP}`, C.ink); bars(k, x + 12, y + 106, 100, 1);
    }
    const [cx, cy] = cur(u); cursor(k, cx, cy);
  });
};
