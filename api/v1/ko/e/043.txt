// 휩 팬 / Whip pan
// 팩: camera (카메라 무빙과 화면 전환) · element 19 · 카테고리: 화면 전환(트랜지션)
// 무엇이 일어나는가: 카메라를 아주 빠르게 옆으로 휘두르고, 블러로 번진 화면 속에서 다음 장면으로 바꾼다.
// 동작 사양:
//   - 연결: 1.7초부터 2.5초까지 0.8초. 앞 0.4초는 앞 숏이 돌아 나가고, 뒤 0.4초는 다음 숏이 돌아 들어와 멈춘다
//   - 전반: 앞 숏의 요 0 → +0.45rad(약 26°), 2차 ease-in-out. 후반: 다음 숏의 요 -0.45rad → 0, 같은 이징
//   - 교체: 진행률 50% 지점에서 바로 바꾼다. 블러가 가장 강한 순간이라 이음매가 보이지 않는다
//   - 블러: ±1.6프레임(30fps 기준) 범위를 9장 렌더링해 평균한다
//   - 카메라: 높이 1.4m. 휘두르는 동안 외에는 고정. 피사체(A는 인물, B는 차와 도시)는 움직이지 않는다
//   - 4.7초마다 A→B와 B→A를 번갈아 실행(왕복 9.4초). 오른쪽 위에 “A → B” / “B → A” 표시
// 쓰임새: 기세를 유지한 채 장소나 화제를 바꿀 때.
// 태그: whip pan, swish pan, motion blur, fast pan, transition, energy
// 크기: 539x208 px · 1주기: 9.4 s
// Needs: kit camera -> https://bbq-man.github.io/motion-prompt-kor/api/v1/kit/camera.js
// 움직이는 견본: https://bbq-man.github.io/motion-prompt-kor/#043
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 19
// 19 ホイップパン（ブレで次の場面へ）
CELLS[18] = shot(abCell((k, w, h, u, A, B) => {
  const t0 = 1.7, t1 = 2.5;
  const at = (kk, s) => {   // s: 0..1 の進み。前半 A を右へ振り、後半 B が左から止まる
    if (s < .5) world(kk, w, h, cam({ y: 1.4, yaw: eIOs(s * 2) * 0.45 }), A === sceneA ? { person: { x: -0.6, z: 7 } } : SB);
    else world(kk, w, h, cam({ y: 1.4, yaw: -(1 - eIOs((s - .5) * 2)) * 0.45 }), B === sceneA ? { person: { x: -0.6, z: 7 } } : SB);
  };
  if (u < t0) return A(k, w, h);
  if (u > t1) return B(k, w, h);
  const s = P(u, t0, t1), ds = 1 / FPS / (t1 - t0);
  mblur(k, at, clamp(s - ds * 1.6), clamp(s + ds * 1.6), 9);
}));
