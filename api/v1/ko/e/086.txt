// 무엇이 일어나는가: 커서가 올라간 링크 아래에 주황색 선이 왼쪽부터 늘어나고, 커서가 벗어나면 선이 오른쪽으로 줄어들며 빠져나간다.
// 동작 사양:
//   - 링크 3개(20px, 굵게, 간격 56). 커서가 왼쪽부터 차례로 약 1초씩 올라간다(히트 영역은 글자 좌우로 6px씩 넓게)
//   - 밑줄: 주황색, 두께 3, 베이스라인 8px 아래
//   - 호버하면 폭 0→100%를 0.3초 동안 왼쪽 끝부터 늘린다(easeOut 3차)
//   - 벗어나면 왼쪽 끝을 0.3초 동안 오른쪽으로 옮겨 선이 오른쪽으로 빠지게 한다(transform-origin을 오른쪽에 둔 scaleX와 같다)
// 쓰임새: 헤더 메뉴나 본문 링크에서 지금 가리키는 곳을 은은하게 보여 주고 싶을 때.
// 태그: underline, hover, link, navigation, menu, text-link, scalex
// 크기: 539x208 px · 1주기: 4.8 s
// Needs: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 움직이는 견본: https://showheyohtaki.github.io/motion-prompt/#086
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 14
// 14 下線アニメーション（リンクにカーソルを乗せると下線が左から伸び、離れると右へ抜ける）
CELLS[13] = (k, w, h, u) => {
  frame(k);
  const LINKS = ['サービス', '実績', '会社概要'], F = `800 20px ${JP}`, GAP = 56, LY = 84, DUR = 0.3;
  const cur = t => path([[0, 120, 150], [0.45, 170, 80], [1.3, 186, 84], [1.55, 268, 80], [2.4, 282, 84], [2.65, 372, 80], [3.5, 390, 84], [3.95, 430, 150], [4.8, 120, 150]], t);
  view(k, () => {
    logo(k, 22, 28); bars(k, 90, 132, 331, 2, C.faint, 6, 12, 0.6);
    k.font = F; const ws = LINKS.map(s => k.measureText(s).width); let x = (VW - ws.reduce((a, b) => a + b, 0) - GAP * (LINKS.length - 1)) / 2;
    LINKS.forEach((s, i) => {
      const x0 = x, x1 = x + ws[i], inside = t => { const [a, b] = cur(t); return a > x0 - 6 && a < x1 + 6 && b > LY - 22 && b < LY + 14; };
      txt(k, s, x0, LY, F, C.ink);
      // 最後に乗った・離れた時刻から 0.3秒（乗る: 左端から伸びる / 離れる: 左端から縮んで右へ抜ける）
      const on = inside(u); let tc = null; for (let j = 0, t = u; j < 600; j++, t -= 1 / 120) if (inside(t - 1 / 120) !== on) { tc = t; break; }
      if (tc !== null) {
        const q = eOut(P(u, tc, tc + DUR)), a = on ? 0 : q, b = on ? q : 1;
        if (b > a) { k.fillStyle = C.acc; k.fillRect(x0 + ws[i] * a, LY + 8, ws[i] * (b - a), 3); }
      }
      x += ws[i] + GAP;
    });
    const [cx, cy] = cur(u); cursor(k, cx, cy);
  });
};
