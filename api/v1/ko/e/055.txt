// 팩: ui (UI 컴포넌트 이름) · element 07 · 카테고리: 열리고 겹치는 컴포넌트
// 무엇이 일어나는가: "상세"를 누르면 오른쪽에 삼각형으로 버튼을 가리키는 흰 작은 창이 뜬다. 창 안의 OK를 누르면 닫힌다.
// 동작 사양:
//   - 4.5초 주기, 카드 542×191. (24, 112)에 파란 테두리 "상세" 버튼(108×36). 커서가 1.0초에 클릭한다.
//   - 표시 정도 a: 1.0–1.28초에 easeOutCubic으로 0→1. 불투명도 a로, 오른쪽에서 6px 미끄러져 들어온다.
//   - 창: 흰색 300×120, 모서리 반경 10, 위치 (156, 62), 1px 테두리, 그림자 blur 22, y 오프셋 6. 왼쪽 변(y 122–138)의 너비 10 삼각형이 버튼을 가리킨다.
//   - 창 안: 굵은 제목 막대, 본문 2줄, 테두리 버튼 70×26과 파란 OK 버튼 72×26.
//   - 닫기: 커서가 3.0초까지 OK (402, 157)로 이동해 3.3초에 클릭한다. a에 (1 − easeInOutCubic(3.3–3.5초))를 곱한다.
// 쓰임새: 버튼 바로 옆에서 짧은 설정이나 확인을 그 자리에서 끝내게 할 때.
// 태그: popover, popup, callout, bubble, flyout, overlay
// 크기: 542x191 px · 1주기: 4.5 s
// Needs: kit ui -> https://bbq-man.github.io/motion-prompt-kor/api/v1/kit/ui.js
// 움직이는 견본: https://bbq-man.github.io/motion-prompt-kor/#055
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 07
// 07 ポップオーバー（Popover）
CELLS[6] = ui((w, h, t) => {
  pageLines(24, 16, 494);
  strokeRR(24, 112, 108, 36, 8, C.blue, 2); text('詳細', 78, 130, 15, C.blue, 600, 'center');
  fillRR(24, 160, 494, 22, 6, C.block);
  const a = eo(seg(t, 1.0, 1.28)) * (1 - eio(seg(t, 3.3, 3.5)));
  if (a > 0) {
    X.save(); X.globalAlpha = a; X.translate(6 * (1 - a), 0);
    const px = 156, py = 62, pw = 300, ph = 120;
    shadowed(() => { fillRR(px, py, pw, ph, 10, '#fff'); }, 22, 6, .2); strokeRR(px, py, pw, ph, 10, C.border, 1);
    X.beginPath(); X.moveTo(px + 1, 122); X.lineTo(px - 10, 130); X.lineTo(px + 1, 138); X.fillStyle = '#fff'; X.fill();
    X.beginPath(); X.moveTo(px, 122); X.lineTo(px - 10, 130); X.lineTo(px, 138); X.strokeStyle = C.border; X.lineWidth = 1; X.stroke();
    bar(px + 18, py + 18, 130, C.ink, 11); bar(px + 18, py + 42, 250); bar(px + 18, py + 60, 200);
    strokeRR(px + 130, py + 82, 70, 26, 6, C.line2, 1.5); fillRR(px + 210, py + 82, 72, 26, 6, C.blue); text('OK', px + 246, py + 95, 14, '#fff', 700, 'center', NUMF);
    X.restore();
  }
  pointer([[.3, w + 30, h + 30], [.95, 84, 134], [1.6, 84, 134], [3.0, 402, 157], [3.5, 402, 157], [4.2, w + 30, h + 30]], [1.0, 3.3], t);
});
