// 교차 편집 / Cross-cutting
// 팩: camera2 (카메라 워크와 화면 전환 2) · element 22 · 카테고리: 화면 전환·편집
// 무엇이 일어나는가: 동시에 벌어지는 두 장면, 달리는 사람과 차가 점점 짧아지는 컷으로 번갈아 나온다.
// 동작 사양:
//   - 두 장면은 하나의 시계를 공유한다. A는 달리는 사람(왼쪽 0.3m, 깊이 15m → 4.5m, 다리 u×12), B는 도시의 차(오른쪽 1m, 깊이 17m → 4.5m). 둘 다 4.7초에 걸쳐 다가온다
//   - 컷: 0 / 1.1 / 2.0 / 2.7 / 3.3 / 3.8 / 4.2 / 4.5초에 A와 B를 스트레이트 컷으로 전환. 샷 길이가 1.1 → 0.9 → 0.7 → 0.6 → 0.5 → 0.4 → 0.3초로 짧아지며 긴장감을 높인다
//   - 한쪽이 화면에 나오는 동안에도 다른 장면은 계속 진행된다(돌아오면 피사체가 그만큼 움직여 있다)
//   - 카메라: A는 높이 1.3m 고정, 전경에 말뚝 4개. B는 높이 1.6m, 피치 +0.06rad 고정, 도시와 해
//   - 하단 바: A(회색)와 B(파랑)가 번갈아 나오는 8구간과 재생 헤드. 마지막 0.3초는 A의 시작으로 크로스페이드
//   - 오른쪽 위에 "SCENE A" / "SCENE B" 표시
// 쓰임새: 추격이나 구조처럼 동시에 벌어지는 일 사이에 긴장감을 쌓을 때.
// 태그: cross-cutting, parallel editing, intercut, parallel action, chase, edit
// 크기: 539x208 px · 1주기: 9.4 s
// Needs: kit camera2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/camera2.js
// 움직이는 견본: https://showheyohtaki.github.io/motion-prompt/#190
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 22
// 22 クロスカッティング: 同時に進む2つの場面（走る人物・走る車）を、だんだん短く交互に見せる
const XC = [0, 1.1, 2.0, 2.7, 3.3, 3.8, 4.2, 4.5, CYC];
CELLS[21] = shot((k, w, h, u) => {
  const r = P(u, CYC - 0.3, CYC);
  const runA = (kk, t) => world(kk, w, h, cam({ y: 1.3 }), { person: { x: -0.3, z: lerp(15, 4.5, t / CYC), walk: t * 12 }, fg: [[-2.2, 3.5], [-2.2, 6.5], [-2.2, 9.5], [-2.2, 12.5]] });
  const carB = (kk, t) => world(kk, w, h, cam({ y: 1.6, pitch: 0.06 }), { city: true, mountains: false, trees: false, sun: [-26, 40, 150], extra: [[1.0, 0, lerp(17, 4.5, t / CYC), CAR[3]]] });
  let i = 0; while (u >= XC[i + 1]) i++;
  const isA = i % 2 === 0;
  blend(k, kk => (isA ? runA : carB)(kk, u), kk => runA(kk, 0), eIOs(r));   // 最後の0.3秒で A の開始状態へ戻す
  const segs = []; for (let j = 0; j < XC.length - 1; j++) segs.push([XC[j], XC[j + 1], j % 2 ? 'B' : 'A', j % 2 ? C.blue : C.g2]);
  strip(k, w, h, segs, u);
  return r > .5 || isA ? 'SCENE A' : 'SCENE B';
});
