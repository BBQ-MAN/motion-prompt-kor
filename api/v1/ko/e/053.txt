// 드롭다운 / Dropdown (select)
// 팩: ui (UI 컴포넌트 이름) · element 05 · 카테고리: 열리고 겹치는 컴포넌트
// 무엇이 일어나는가: 입력란을 누르면 아래로 선택지 4개가 펼쳐지고, "나고야"를 고르면 목록이 닫히며 입력란에 들어간다.
// 동작 사양:
//   - 4.5초 주기, 카드 542×191. 260×38 입력란(모서리 반경 8)에 플레이스홀더와 셰브런. 커서가 1.0초에 클릭한다.
//   - 열림 정도: 1.0–1.25초에 easeOutCubic으로 0→1. 목록(흰색 260×112, 그림자 blur 18)이 6px 내려오며 페이드인된다. 셰브런은 180° 돌고 입력란 테두리는 파란 2px가 된다.
//   - 선택지 4개가 26px 간격(화면은 일본어로 도쿄, 오사카, 나고야, 후쿠오카). 커서가 내려가면서 1.55초부터 오사카, 1.95초부터 나고야 줄이 연한 파랑으로 강조된다.
//   - 2.3초에 나고야 클릭: 목록은 2.3–2.45초에 (1 − easeInOutCubic)으로 닫히고, 플레이스홀더는 2.2–2.3초에 사라지며 "나고야"가 2.32–2.42초에 페이드인된다.
//   - 리셋: 나고야가 3.9–4.08초에 사라진 뒤 플레이스홀더가 4.1–4.3초에 돌아온다. 두 텍스트는 겹치지 않는다.
// 쓰임새: 폼에서 정해진 후보 중 하나를 고르게 할 때.
// 태그: dropdown, dropdown-menu, select, picker, combobox, options, form
// 크기: 542x191 px · 1주기: 4.5 s
// Needs: kit ui -> https://bbq-man.github.io/motion-prompt-kor/api/v1/kit/ui.js
// 움직이는 견본: https://bbq-man.github.io/motion-prompt-kor/#053
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 05
// 05 ドロップダウン（Dropdown menu）
CELLS[4] = ui((w, h, t) => {
  bar(24, 14, 90, C.line2, 10);
  const open = eo(seg(t, 1.0, 1.25)) * (1 - eio(seg(t, 2.3, 2.45)));
  const chosen = seg(t, 2.32, 2.42) * (1 - seg(t, 3.9, 4.08));
  const ph = (1 - seg(t, 2.2, 2.3)) + seg(t, 4.1, 4.3); // プレースホルダーは先に消し、名前が消えてから戻す（重ねない）
  strokeRR(24, 34, 260, 38, 8, open > .1 || (chosen > 0 && t < 2.9) ? C.blue : C.line2, open > .1 ? 2 : 1.5);
  if (ph > 0) { X.globalAlpha = clamp(ph); text('選択してください', 40, 53, 15, '#9a9896', 500); X.globalAlpha = 1; }
  if (chosen > 0) { X.globalAlpha = chosen; text('名古屋', 40, 53, 16, C.ink, 600); X.globalAlpha = 1; }
  chevron(262, 52, 5, Math.PI * open, '#777');
  fillRR(310, 34, 208, 70, 8, C.block); bar(310, 120, 208); bar(310, 138, 160); bar(24, 150, 200); bar(24, 168, 150);
  if (open > 0) {
    X.save(); X.globalAlpha = open; const oy = 78 - 6 * (1 - open);
    shadowed(() => fillRR(24, oy, 260, 112, 8, '#fff'), 18, 6, .18); strokeRR(24, oy, 260, 112, 8, C.border, 1);
    const hi = seg(t, 1.55, 1.75) > 0 ? (t < 1.95 ? 1 : 2) : -1;
    ['東京', '大阪', '名古屋', '福岡'].forEach((s, i) => { const yy = oy + 4 + i * 26; if (i === hi + (hi === 1 ? 0 : 0) && hi >= 0) fillRR(28, yy, 252, 26, 5, C.lb); text(s, 40, yy + 13, 15, i === hi ? C.blue : '#444', i === hi ? 600 : 500); });
    X.restore();
  }
  pointer([[.3, w + 30, h + 30], [.95, 200, 56], [1.25, 200, 56], [1.6, 170, 121], [2.0, 170, 147], [2.45, 170, 147], [3.2, 170, 147], [4.2, w + 30, h + 30]], [1.0, 2.3], t);
});
