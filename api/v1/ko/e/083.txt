// 커서 팔로워 / Cursor follower
// 팩: lp (랜딩 페이지·웹사이트 연출) · element 11 · 카테고리: 호버·클릭 인터랙션
// 무엇이 일어나는가: 커서가 페이지 위를 돌아다니면 주황색 원이 조금 늦게 그 뒤를 따라간다.
// 동작 사양:
//   - 커서 경로: x = 255 + 175·sin(2πt/4.8), y = 86 + 46·sin(4πt/4.8)(옆으로 눕힌 8자 모양, 4.8초에 한 바퀴)
//   - 원: 반지름 17, 주황색, 불투명도 0.88. 커서 아래에 그린다
//   - 따라가는 방식: 매 프레임(60fps) 남은 거리의 12%씩 다가간다(x += (커서 x − x) × 0.12)
//   - 화살표 커서는 그대로 보인다
// 쓰임새: 브랜드 사이트나 포트폴리오에 직접 만지는 느낌과 재미를 더하고 싶을 때.
// 태그: cursor, custom-cursor, cursor-follower, mouse-follow, lerp, trail, interaction
// 크기: 539x208 px · 1주기: 4.8 s
// Needs: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 움직이는 견본: https://showheyohtaki.github.io/motion-prompt/#083
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 11
// 11 マウスストーカー（カーソルのあとを、丸が少し遅れて追いかける）
CELLS[10] = (k, w, h, u) => {
  frame(k);
  const cur = t => [255 + 175 * Math.sin(2 * Math.PI * t / CYC), 86 + 46 * Math.sin(4 * Math.PI * t / CYC)];
  // 毎フレーム（60fps）残りの距離の 12% ずつ近づく
  let t = u - 2, [fx, fy] = cur(t);
  for (let i = 0; i < 120; i++) { t += 1 / 60; const [tx, ty] = cur(t); fx += (tx - fx) * 0.12; fy += (ty - fy) * 0.12; }
  view(k, () => {
    txt(k, 'CREATIVE STUDIO', VW / 2, 74, `800 34px ${EN}`, C.ink, 'center', 'alphabetic', '3px');
    bars(k, VW / 2 - 120, 96, 240, 2, C.faint, 6, 12, 0.7);
    btn(k, VW / 2 - 50, 128, 100, 24, '作品を見る', C.ink);
    k.fillStyle = C.acc; k.globalAlpha = 0.88; k.beginPath(); k.arc(fx, fy, 17, 0, 7); k.fill(); k.globalAlpha = 1;
    const [cx, cy] = cur(u); cursor(k, cx, cy);
  });
};
