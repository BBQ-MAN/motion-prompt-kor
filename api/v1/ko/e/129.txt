// 팩: motion2 (모션 그래픽 연출 2탄) · element 09 · 카테고리: 텍스트·도형 연출
// 무엇이 일어나는가: 무작위 문자가 계속 바뀌다가 왼쪽부터 차례로 올바른 단어로 확정된다.
// 동작 사양:
//   - 앞 2.35초는 "HELLO", 뒤는 "WORLD". 5글자 고정폭 글꼴(92px).
//   - 확정 전 칸: A–Z(I와 O 제외), 0–9, #$%&@? 중에서 무작위로 골라 4.7초에 76회(약 0.062초마다) 바꾼다. 민트색, 불투명도 0.75.
//   - 왼쪽부터 확정: i번째 글자는 0.45 + 0.18i초(마지막은 1.17초). 확정 순간부터 0.22초는 노랑, 이후 흰색. 크기 1 + 0.2·e^(−14t)로 안정된다.
//   - 1.85초부터 오른쪽부터 차례로 다시 무작위로 돌아간다(i번째 글자는 1.85 + 0.08 × (4 − i)초).
//   - 각 글자 아래에 가로 막대: 확정된 글자는 파랑, 다음에 확정될 글자는 노랑, 나머지는 남색.
// 쓰임새: 제목, 숫자, 코드명이 등장할 때. 해독, 해킹, 테크 분위기를 내고 싶을 때.
// 태그: text-scramble, scramble, decode, shuffle, random, text, hacker
// 크기: 539x208 px · 1주기: 4.7 s
// Needs: kit motion2 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/motion2.js
// 움직이는 견본: https://showheyohtaki.github.io/motion-prompt/#129
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 09
// 09 テキストスクランブル（ランダムな文字が入れ替わり続け、左から順に正しい単語に決まる）
const SCR_GL = 'ABCDEFGHJKLMNPQRSTUVWXYZ0123456789#$%&@?';
CELLS[8] = (k, w, h, u) => {
  u = snapU(u);
  stage(k);
  const T = CYC / 2, n = Math.floor(u / T), lu = u - n * T, word = n ? 'WORLD' : 'HELLO', N = 5;
  const f = `700 92px Menlo, "SF Mono", monospace`; k.font = f;
  const cw = k.measureText('M').width + 16, x0 = w / 2 - cw * (N - 1) / 2, Y = h / 2 - 8;
  const step = Math.floor(u / (CYC / 76));
  for (let i = 0; i < N; i++) {
    const tl = 0.45 + i * 0.18, tu = 1.85 + (N - 1 - i) * 0.08, locked = lu >= tl && lu < tu;
    let ch, col, sc = 1;
    if (locked) { ch = word[i]; const age = lu - tl; col = age < 0.22 ? C.yel : '#fff'; sc = 1 + 0.2 * Math.exp(-age * 14); }
    else { ch = SCR_GL[Math.floor(H01(step * 131 + i * 977 + n * 31) * SCR_GL.length)]; col = C.mint; }
    k.save(); k.translate(x0 + i * cw, Y); k.scale(sc, sc); k.globalAlpha = locked ? 1 : 0.75; txt(k, ch, 0, 0, f, col, 'center', 'middle'); k.restore();
    const next = !locked && lu < tl && (i === 0 || lu >= 0.45 + (i - 1) * 0.18);
    k.fillStyle = locked ? C.blue : next ? C.yel : C.navy3; rr(k, x0 + i * cw - cw / 2 + 8, Y + 58, cw - 16, 8, 4); k.fill();
  }
};
