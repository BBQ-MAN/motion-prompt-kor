// 무한 스크롤 / Infinite scroll
// 팩: lp (랜딩 페이지·웹사이트 연출) · element 19 · 카테고리: 로딩·페이지 이동·반복 연출
// 무엇이 일어나는가: 목록을 아래로 스크롤해 끝에 이르면 스피너가 돌고, 새 항목 3개가 추가되어 계속 읽을 수 있다.
// 동작 사양:
//   - 항목(331×44, 모서리 반경 8, 간격 52) 목록. 마지막 항목 아래에 스피너(반지름 10, 0.8초에 1회전)
//   - 스크롤 양: 0.25~0.85초에 38→116px, 1.15~1.75초에 116→194px(easeInOut 3차). 목록 끝이 보이는 곳에서 멈춘다
//   - 2.6초에 3개를 추가: 각 항목이 0.35초 동안 불투명도 0→1, 10px 아래에서 올라온다(easeOut 3차). 스피너는 새 끝으로 옮겨진다
//   - 한 주기에 항목 3개(156px)만큼 진행한다. 내용은 3개마다 반복되므로 주기의 끝과 시작이 같아 보인다
// 쓰임새: SNS 피드나 공지 목록처럼 페이지를 넘기지 않고 계속 읽게 하고 싶을 때.
// 태그: infinite-scroll, endless-scroll, load-more, feed, pagination, spinner, list
// 크기: 539x208 px · 1주기: 4.8 s
// Needs: kit lp -> https://bbq-man.github.io/motion-prompt-kor/api/v1/kit/lp.js
// 움직이는 견본: https://bbq-man.github.io/motion-prompt-kor/#091
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 19
// 19 無限スクロール（いちばん下まで来ると読み込み表示が出て、続きが足される）
CELLS[18] = (k, w, h, u) => {
  frame(k);
  // 1周期で 3件（156px）ぶん進む。周期の終わりは始まりと同じ見た目（内容は3件ごとにくり返す）
  const PITCH = 52, BATCH = 3, L0 = 6, TL = 2.6;
  const y = steps([[0.25, 38], [0.85, 116], [1.15, 116], [1.75, 194], [4.8, 194]], u);
  const n = u < TL ? L0 : L0 + BATCH, end = 10 + n * PITCH;
  view(k, () => {
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH);
    k.save(); k.translate(0, -y);
    for (let i = 0; i < n; i++) {
      const top = 10 + i * PITCH, v = i % 3, fresh = i >= L0 ? eOut(P(u, TL, TL + 0.35)) : 1;
      if (top - y > VH || top + 44 - y < 0) continue;
      k.save(); k.globalAlpha = fresh; k.translate(0, 10 * (1 - fresh));
      k.fillStyle = '#fff'; rr(k, 90, top, 331, 44, 8); k.fill(); pic(k, 98, top + 6, 48, 32, v, 5);
      bars(k, 158, top + 12, [180, 140, 200][v], 1, C.ink, 7); bars(k, 158, top + 26, [120, 150, 100][v], 1, C.faint, 6);
      k.restore();
    }
    spinner(k, VW / 2, end + 18, 10, u);
    k.restore();
  });
};
