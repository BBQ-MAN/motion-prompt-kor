// 스크롤 줌 / Zoom on scroll
// 팩: lp2 (랜딩 페이지·웹사이트 연출 2탄) · element 08 · 카테고리: 스크롤 연출
// 무엇이 일어나는가: 페이지를 스크롤하면 가운데 작은 사진이 창 전체로 커지고, 주변 텍스트는 사라지며 사진 위에 헤드라인이 나타난다.
// 동작 사양:
//   - 511 × 168 뷰포트. 스크롤 진행도 p: 0.3–2.3초에 0→1, 3.0–4.4초에 1→0(cubic ease-in-out). 1회 4.7초(팩 주기 9.4초에 2회).
//   - 사진 프레임: 가운데 155 × 90(x = 178, y = 58, radius 8)에서 창 전체 511 × 168(radius 0)까지. 위치, 폭, 높이, radius가 p에 비례해 변한다.
//   - 주변 제목과 텍스트는 p가 0→0.45인 동안 페이드아웃된다(opacity 1 − 2.2p).
//   - p가 0.7→1인 동안 사진 중앙에 흰색 헤드라인(26 px, 자간 2 px)이 페이드인된다. 스크롤바 썸은 스크롤을 따라간다.
// 쓰임새: 여행, 부동산, 브랜드 사이트에서 방문자를 사진 속으로 끌어들일 때.
// 태그: zoom, scroll, image, scale, scroll-driven, hero, fullscreen
// 크기: 539x208 px · 1주기: 9.4 s
// Needs: kit lp2 -> https://bbq-man.github.io/motion-prompt-kor/api/v1/kit/lp2.js
// 움직이는 견본: https://bbq-man.github.io/motion-prompt-kor/#152
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 08
// 08 スクロールで拡大（スクロールに合わせて、小さな写真が窓いっぱいまで広がる）
CELLS[7] = (k, w, h, u) => {
  u = snapU(u);
  frame(k);
  const p = steps([[0.3, 0], [2.3, 1], [3.0, 1], [4.4, 0]], u);
  const x = lerp(178, 0, p), y = lerp(58, 0, p), ww = lerp(155, VW, p), hh = lerp(90, VH, p);
  view(k, () => {
    const a = 1 - clamp(p * 2.2);
    k.save(); k.globalAlpha = a;
    txt(k, '景色の中へ', VW / 2, 38, `800 19px ${JP}`, C.ink, 'center');
    bars(k, 34, 66, 120, 4, C.faint, 6, 14); bars(k, 357, 66, 120, 4, C.faint, 6, 14);
    k.restore();
    pic(k, x, y, ww, hh, 0, lerp(8, 0, p));
    const b = clamp((p - 0.7) / 0.3);
    if (b > 0) { k.save(); k.globalAlpha = b; txt(k, '絶景へようこそ', VW / 2, VH / 2 + 2, `800 26px ${JP}`, '#fff', 'center', 'middle', '2px'); k.restore(); }
    sbar(k, p * 0.6, 0.4);
  });
};
