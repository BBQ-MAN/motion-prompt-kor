// FPV 드론 숏 / FPV drone shot
// 팩: camera (카메라 무빙과 화면 전환) · element 11 · 카테고리: 촬영 기법과 시간 연출
// 무엇이 일어나는가: 가로수 사이를 낮고 빠르게 비행하며 기체를 좌우로 기울여 지그재그로 나아가는 1인칭 시점.
// 동작 사양:
//   - 카메라: 앞으로 등속 14m/s(50km/h로 표시), 이징 없이 계속 전진. 피치는 -0.05rad로 고정
//   - 지그재그(T = 2π×u/4.7, 4.7초에 2회 왕복): 가로 위치 0.6·sin(2T-0.4) m, 높이 1.4 + 0.5·sin(2T) m
//   - 뱅크: 롤 0.22·sin(2T+0.6) rad(약 ±12.6°), 요 0.05·sin(2T) rad(약 ±2.9°)
//   - 피사체 없음. 좌우 3.6m 위치에 높이 4.2m 나무를 8m 간격으로 세우고, 좌우 열은 4m 엇갈리게 배치. 나무가 차례로 카메라 쪽으로 밀려온다
//   - 스피드 라인: 중심에서 방사형으로 10개, 길이 약 32~76px(옆쪽일수록 길다), 반투명 회색. 바깥으로 흐르며 순환한다
//   - 정지와 복귀 크로스페이드 없음(4.7초마다 다시 시작할 때 나무 배열의 위상이 어긋난다)
// 쓰임새: 속도감과 몰입감을 줄 때, 또는 장소 안을 한 번에 통과하며 보여 줄 때.
// 태그: fpv, drone, fly through, first person, banking, speed, aerial
// 크기: 539x208 px · 1주기: 9.4 s
// Needs: kit camera -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/camera.js
// 움직이는 견본: https://showheyohtaki.github.io/motion-prompt/#035
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D(호출하는 쪽에서 지우기와 save/restore 완료), w·h = 크기, u = 초(0 이상 1주기 미만). u만으로 그림이 정해진다.
// 여기부터 요소의 코드. CELLS와 상수·보조 함수는 kit에 있다.

// @element 11
// 11 FPV
CELLS[10] = shot((k, w, h, u) => {
  const sp = 14, z = u * sp, T = u * 2 * Math.PI / CYC;
  const c = cam({ z, y: 1.4 + 0.5 * Math.sin(T * 2), roll: 0.22 * Math.sin(T * 2 + 0.6), yaw: 0.05 * Math.sin(T * 2), x: 0.6 * Math.sin(T * 2 - 0.4), pitch: -0.05 });
  const ex = [];
  const L = 8, base = Math.floor(z / L) * L;
  for (let i = 0; i < 12; i++) { const zz = base + i * L; ex.push([-3.6, 0, zz, (kk, s) => tree(kk, s, 4.2)]); ex.push([3.6, 0, zz + L / 2, (kk, s) => tree(kk, s, 4.2, C.blueXL)]); }
  world(k, w, h, c, {
    buildings: false, trees: false, extra: ex, gridX: 20,
    after: (kk) => {   // スピード線
      kk.strokeStyle = 'rgba(150,150,155,.45)'; kk.lineWidth = 2;
      for (let i = 0; i < 10; i++) { const a = i / 10 * Math.PI * 2 + 0.3, ph = mod(u * 2.2 + i * 0.37, 1); const r0 = 120 + ph * 160; kk.beginPath(); kk.moveTo(Math.cos(a) * r0 * 1.9, Math.sin(a) * r0 * 0.8); kk.lineTo(Math.cos(a) * (r0 + 40) * 1.9, Math.sin(a) * (r0 + 40) * 0.8); kk.stroke(); }
    }
  });
  return `${Math.round(sp * 3.6)}km/h`;
});
